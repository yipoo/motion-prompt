Motion Prompt #120 · Double exposure
The background turns white and the landscape of hills, house, trees and sun appears, scaled down, only inside the shape of the bust.

- Scene 3: a bust on the left (head centre x 140, y 82, radius 28 px) and hills, a house, trees and the sun (x 432, y 54, radius 15 px) on the right.
- Background: white #FFFFFF outside the bust
- Inside the bust: the same landscape at 0.7×, placed so the sun sits at the top right of the head and the ridge of the hills runs across the neck
- Inside colours: sky #D3DBE6, hills #9AA8BD and #76869F, ground #55647E, trees #3F4C66
- Transition: the double exposure is laid over the original picture at the effect amount
- Timing: the effect comes in from 0.4 s to 1.4 s, holds until 4.0 s and is gone again by 4.8 s (one pass = 5 s, cubic ease-in-out (about cubic-bezier(0.65,0,0.35,1)))

Canvas: 539×208; cycle: 5s.

  // @element 24
    CELLS[23] = cell((k, u) => {
    const a = env(u);
    const DX = Object.assign({}, BASE, {
      sky: "#D3DBE6",
      sky2: "#D3DBE6",
      far: "#9AA8BD",
      mid: "#76869F",
      ground: "#55647E",
      bld: "#E8ECF2",
      edge: "#55647E",
      win: "#9AA8BD",
      winOff: "#9AA8BD",
      tree: "#3F4C66",
      trunk: "#3F4C66"
    });
    const draw = kk => {
      kk.fillStyle = "#FFFFFF";
      kk.fillRect(0, 0, CW, CH);
      kk.save();
      bustPath(kk, QX, QY, QR);
      kk.clip();
      kk.translate(QX, QY);
      kk.scale(.7, .7);
      kk.translate(-415, -77);
      kk.fillStyle = DX.sky;
      kk.fillRect(0, -200, CW * 2, CH * 3);
      stageLand(kk, DX, {
        person: false
      });
      kk.fillStyle = DX.ground;
      kk.fillRect(-CW, 200, CW * 3, CH * 3);
      kk.restore();
    };
    if (a >= .999) {
      draw(k);
      return;
    }
    stageLand(k, BASE);
    if (a < .002) return;
    const [L, kl] = layer("dx");
    draw(kl);
    k.save();
    k.globalAlpha = a;
    k.drawImage(L, 0, 0);
    k.restore();
  });
