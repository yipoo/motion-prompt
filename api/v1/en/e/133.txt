Motion Prompt #133 · Text mask
Heavy letters are cut out so that a moving seascape shows only inside them; near the end of the cycle the whole picture appears.

- "OCEAN" (weight 900, system-ui, 2 px letter spacing). The font size starts at 172 px and drops 2 px at a time until the word is at most 500 px wide.
- The scene inside keeps moving: coral sky, a yellow sun bobbing ±6 px, blue sea below y = 116 with four white wavy lines drifting 240 px per 4.7 s in alternating directions, a sailboat crossing once per 4.7 s, clouds drifting 60 px.
- Method: draw the word on an offscreen canvas, then composite the scene onto it with source-in, so the scene remains only inside the letters. A white 4 px outline around the letters.
- Outside the letters the scene shows at 12% opacity; it rises to 100% at 3.45–3.85 s and returns to 12% at 4.25–4.65 s (smoothstep).

Canvas: 539×208; cycle: 4.7s.

  function seaPattern(g, w, h, u) {
    g.fillStyle = C.coral;
    g.fillRect(0, 0, w, h);
    g.fillStyle = C.yel;
    g.beginPath();
    g.arc(w * .6, 116 + 6 * Math.sin(2 * Math.PI * u / CYC), 60, 0, 7);
    g.fill();
    g.fillStyle = C.blue;
    g.fillRect(0, 116, w, h - 116);
    const off = u / CYC * 240;
    g.strokeStyle = "#fff";
    g.lineWidth = 7;
    g.lineCap = "round";
    for (let j = 0; j < 4; j++) {
      const y = 128 + j * 22, d = j % 2 ? -1 : 1;
      g.beginPath();
      for (let x = -20; x <= w + 20; x += 6) {
        const yy = y + 5 * Math.sin((x + off * d + j * 40) * 2 * Math.PI / 120);
        x === -20 ? g.moveTo(x, yy) : g.lineTo(x, yy);
      }
      g.stroke();
    }
    const bx = mod(u / CYC * (w + 160), w + 160) - 80;
    g.fillStyle = "#fff";
    g.beginPath();
    g.moveTo(bx, 46);
    g.lineTo(bx, 100);
    g.lineTo(bx + 40, 100);
    g.closePath();
    g.fill();
    g.fillStyle = C.navy;
    g.beginPath();
    g.moveTo(bx - 42, 104);
    g.lineTo(bx + 52, 104);
    g.lineTo(bx + 38, 122);
    g.lineTo(bx - 30, 122);
    g.closePath();
    g.fill();
    g.fillStyle = "#fff";
    [ [ 60, 40 ], [ 320, 30 ], [ 470, 56 ] ].forEach(([x, y]) => {
      const xx = mod(x + u / CYC * 60, w + 80) - 40;
      g.beginPath();
      g.ellipse(xx, y, 30, 11, 0, 0, 7);
      g.fill();
    });
  }
