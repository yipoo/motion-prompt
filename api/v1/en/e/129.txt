Motion Prompt #129 · Text scramble
Random characters keep changing, then lock into the right word from left to right.

- "HELLO" in the first 2.35 s, "WORLD" in the second. Five monospaced letters (92 px).
- Unsettled slots show random characters from A–Z (without I and O), 0–9 and #$%&@?, changing 76 times per 4.7 s (about every 0.062 s), mint at 75% opacity.
- Slots lock from left to right: slot i at 0.45 + 0.18i s (the last at 1.17 s). A newly locked letter is yellow for 0.22 s, then white, settling with scale 1 + 0.2·e^(−14t).
- From 1.85 s the slots return to random from right to left (slot i at 1.85 + 0.08 × (4 − i) s).
- A bar under each slot: blue when locked, yellow for the next one to lock, navy otherwise.

Canvas: 539×208; cycle: 4.7s.

  // @element 09
    const SCR_GL = "ABCDEFGHJKLMNPQRSTUVWXYZ0123456789#$%&@?";
  CELLS[8] = (k, w, h, u) => {
    u = snapU(u);
    stage(k);
    const T = CYC / 2, n = Math.floor(u / T), lu = u - n * T, word = n ? MotionLocale.text("WORLD") : MotionLocale.text("HELLO"), N = [ ...word ].length;
    const f = `700 92px Menlo, "SF Mono", monospace`;
    k.font = f;
    const cw = k.measureText("M").width + 16, x0 = w / 2 - cw * (N - 1) / 2, Y = h / 2 - 8;
    const step = Math.floor(u / (CYC / 76));
    for (let i = 0; i < N; i++) {
      const tl = .45 + i * .18, tu = 1.85 + (N - 1 - i) * .08, locked = lu >= tl && lu < tu;
      let ch, col, sc = 1;
      if (locked) {
        ch = word[i];
        const age = lu - tl;
        col = age < .22 ? C.yel : "#fff";
        sc = 1 + .2 * Math.exp(-age * 14);
      } else {
        ch = SCR_GL[Math.floor(H01(step * 131 + i * 977 + n * 31) * SCR_GL.length)];
        col = C.mint;
      }
      k.save();
      k.translate(x0 + i * cw, Y);
      k.scale(sc, sc);
      k.globalAlpha = locked ? 1 : .75;
      txt(k, ch, 0, 0, f, col, "center", "middle");
      k.restore();
      const next = !locked && lu < tl && (i === 0 || lu >= .45 + (i - 1) * .18);
      k.fillStyle = locked ? C.blue : next ? C.yel : C.navy3;
      rr(k, x0 + i * cw - cw / 2 + 8, Y + 58, cw - 16, 8, 4);
      k.fill();
    }
  };
