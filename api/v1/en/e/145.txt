Motion Prompt #145 · Timeline on scroll
As the page scrolls down, an orange line grows down the timeline; each dot it reaches fills in and its year and event slide in from the right. Scrolling back up reverses it.

- A 511 × 168 browser viewport. Scroll: 0→190 px over 0.3–2.3 s, 190→0 over 3.0–4.3 s (cubic ease-in-out). One pass takes 4.7 s (twice per 9.4 s pack cycle).
- The timeline line is at x = 168: an orange line grows over a gray track (width 3). Its tip sits at scroll + 126 px, so always 126 px from the top of the viewport (42 px from the bottom).
- Five items (2018–2026), with dots (white, radius 7.5, ring width 2.5) at page y = 72, 126, 180, 234 and 288.
- Over the 22 px of line growth after the tip passes a dot (cubic ease-out): the ring turns orange, an inner dot grows from radius 0 to 4, and the year (18 px, orange) and event (15 px) fade in while sliding 16 px from the right.
- Scrolling back shrinks the line and hides the items by the same formula. The scrollbar thumb on the right follows the scroll.

Canvas: 539×208; cycle: 9.4s.

  // @element 01
    CELLS[0] = (k, w, h, u) => {
    u = snapU(u);
    frame(k);
    const y = steps([ [ .3, 0 ], [ 2.3, 190 ], [ 3, 190 ], [ 4.3, 0 ] ], u);
    const ROWS = [ [ 72, "2018", MotionLocale.text("copy_073") ], [ 126, "2020", MotionLocale.text("copy_074") ], [ 180, "2022", MotionLocale.text("copy_076") ], [ 234, "2024", MotionLocale.text("copy_075") ], [ 288, "2026", MotionLocale.text("copy_077") ] ];
    const LX = 168, L = y + 126;
    view(k, () => {
      k.save();
      k.translate(0, -y);
      txt(k, MotionLocale.text("HISTORY"), 34, 34, `800 11px ${EN}`, C.acc, "left", "alphabetic", "2px");
      txt(k, MotionLocale.text("copy_072"), 34, 60, `800 20px ${FONT}`, C.ink);
      pic(k, 34, 76, 100, 62, 0, 6);
      bars(k, 34, 148, 90, 2);
      k.fillStyle = C.faint;
      rr(k, LX - 1.5, 52, 3, 248, 1.5);
      k.fill();
      k.fillStyle = C.acc;
      rr(k, LX - 1.5, 52, 3, clamp(L - 52, 0, 248), 1.5);
      k.fill();
      for (const [iy, yr, lab] of ROWS) {
        const q = eOut(clamp((L - iy) / 22));
        k.fillStyle = "#fff";
        k.beginPath();
        k.arc(LX, iy, 7.5, 0, 7);
        k.fill();
        k.lineWidth = 2.5;
        k.strokeStyle = q > 0 ? C.acc : C.faint;
        k.stroke();
        if (q > 0) {
          k.fillStyle = C.acc;
          k.beginPath();
          k.arc(LX, iy, 4 * q, 0, 7);
          k.fill();
        }
        k.save();
        k.globalAlpha = q;
        k.translate(16 * (1 - q), 0);
        txt(k, yr, LX + 22, iy + 6, `800 18px ${EN}`, C.acc);
        txt(k, lab, LX + 82, iy + 5, `700 15px ${FONT}`, C.ink);
        bars(k, LX + 82, iy + 15, 190, 1, C.faint, 5);
        k.restore();
      }
      k.restore();
      sbar(k, y / 190, .45);
    });
  };
