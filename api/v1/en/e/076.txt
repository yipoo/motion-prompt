Motion Prompt #076 · Pinning
A navy section stops when it fills the view; further scrolling steps the number and photo through 1→2→3, then the section scrolls on.

- Scroll: 0→500 px at 0.3–3.0 s, 500→0 at 3.5–4.5 s (cubic ease-in-out).
- The section (168 tall, navy) reaches the top at 110 px of scroll and stays there for the next 300 px (a GSAP ScrollTrigger pin).
- Progress while pinned: p = (scroll − 110) / 300. A vertical line (124 long) on the left fills in orange with p.
- At p = 1/3 and 2/3 (±0.07, cubic ease-in-out) it moves to the next step: the number (46 px) rolls up 52 px and the photo (226 × 124) slides up 124 px.
- After 300 px the pin is released and the section scrolls up with the white section below it.

Canvas: 539×208; cycle: 4.8s.

  // @element 04
    CELLS[3] = (k, w, h, u) => {
    frame(k);
    const y = steps([ [ .3, 0 ], [ 3, 500 ], [ 3.5, 500 ], [ 4.5, 0 ] ], u);
    const PIN0 = 110, PD = 300;
    const top = y < PIN0 ? PIN0 - y : y < PIN0 + PD ? 0 : PIN0 + PD - y, p = clamp((y - PIN0) / PD);
    const f = eIO(P(p, 1 / 3 - .07, 1 / 3 + .07)) + eIO(P(p, 2 / 3 - .07, 2 / 3 + .07));
    view(k, () => {
      txt(k, MotionLocale.text("copy_155"), 34, 56 - y, `800 22px ${FONT}`, C.ink);
      bars(k, 34, 72 - y, 240, 2);
      k.save();
      k.translate(0, top);
      k.fillStyle = C.ink;
      k.fillRect(0, 0, VW, VH);
      txt(k, MotionLocale.text("STEP"), 34, 40, `700 12px ${EN}`, "rgba(255,255,255,.6)", "left", "alphabetic", "2px");
      k.save();
      k.beginPath();
      k.rect(30, 46, 120, 52);
      k.clip();
      for (let i = 0; i < 3; i++) txt(k, "0" + (i + 1), 32, 90 + (i - f) * 52, `800 46px ${EN}`, "#fff");
      k.restore();
      bars(k, 34, 110, 150, 3, "rgba(255,255,255,.28)");
      k.fillStyle = "rgba(255,255,255,.16)";
      rr(k, 232, 22, 4, 124, 2);
      k.fill();
      k.fillStyle = C.acc;
      rr(k, 232, 22, 4, 124 * p, 2);
      k.fill();
      k.save();
      rr(k, 256, 22, 226, 124, 8);
      k.clip();
      for (let i = 0; i < 3; i++) pic(k, 256, 22 + (i - f) * 124, 226, 124, i, 0);
      k.restore();
      k.fillStyle = "#fff";
      k.fillRect(0, VH, VW, 200);
      bars(k, 34, VH + 30, 300, 3);
      k.restore();
      sbar(k, y / 500, .3);
    });
  };
