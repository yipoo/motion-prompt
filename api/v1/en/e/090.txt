Motion Prompt #090 · Lazy loading
As grey placeholders scroll into view, a spinner turns in each, and the images fade in as they arrive. The top row is loaded from the start.

- A 3 × 3 grid of 151 × 92 placeholders; the first row is shown from the start. Scroll: 0→110 px at 0.35–1.1 s, 110→225 px at 1.7–2.5 s, 225→0 at 3.5–4.5 s (cubic ease-in-out).
- Loading starts when a placeholder's top edge passes above the bottom of the viewport (168 px), with each column 0.12 s later than the one to its left.
- For 0.45 s a spinner (radius 11, one turn per 0.8 s) turns in the placeholder's centre.
- Then the image fades in 0→1 over 0.35 s (cubic ease-out). Unloaded placeholders show an image icon.
- Placeholders that leave the bottom of the view are reset out of sight.

Canvas: 539×208; cycle: 4.8s.

  // @element 18
    CELLS[17] = (k, w, h, u) => {
    frame(k);
    const SY = t => steps([ [ .35, 0 ], [ 1.1, 110 ], [ 1.7, 110 ], [ 2.5, 225 ], [ 3.5, 225 ], [ 4.5, 0 ] ], t), y = SY(u);
    const COLX = [ 15, 180, 345 ], IW = 151, IH = 92, ROWS = [ 12, 168, 276 ], WAIT = .45, FADE = .35;
    view(k, () => {
      k.save();
      k.translate(0, -y);
      txt(k, MotionLocale.text("copy_154"), 15, 140, `800 18px ${FONT}`, C.ink);
      bars(k, 120, 128, 200, 1);
      ROWS.forEach((top, r) => COLX.forEach((x, c) => {
        const t0 = r === 0 ? -1e9 : enteredAt(SY, top, VH, u), age = t0 === null ? -1 : u - t0 - c * .12;
        k.fillStyle = C.skel;
        rr(k, x, top, IW, IH, 6);
        k.fill();
        if (age < WAIT + FADE) {
          k.save();
          k.translate(x + IW / 2, top + IH / 2);
          k.strokeStyle = "#B9C1CF";
          k.lineWidth = 2;
          k.lineJoin = "round";
          if (age >= 0 && age < WAIT) spinner(k, 0, 0, 11, u, C.blue); else {
            rr(k, -15, -11, 30, 22, 3);
            k.stroke();
            k.beginPath();
            k.moveTo(-11, 7);
            k.lineTo(-3, -1);
            k.lineTo(3, 5);
            k.lineTo(7, 1);
            k.lineTo(11, 6);
            k.stroke();
          }
          k.restore();
        }
        if (age >= WAIT) {
          k.save();
          k.globalAlpha = eOut(P(age, WAIT, WAIT + FADE));
          pic(k, x, top, IW, IH, (r + c) % 3, 6);
          k.restore();
        }
      }));
      k.restore();
      sbar(k, y / 225, .45);
    });
  };
