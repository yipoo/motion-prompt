Motion Prompt #082 · Image zoom on hover
The photo under the cursor slowly enlarges while its frame stays the same size, and shrinks back when the cursor leaves.

- Two photo frames, 222 × 118 (radius 8). The cursor rests on the left one, then the right one, for about 1.5 s each.
- On hover the content scales 1→1.15 from the centre over 0.5 s (cubic ease-out); on leave it returns over 0.5 s.
- Anything outside the frame is clipped (overflow: hidden). The frame and the caption do not move.

Canvas: 539×208; cycle: 4.8s.

Dependencies: api/v1/kit/lp.js + studio/locale.js + assets/fonts/NotoSansSC.woff2
Set document.documentElement.lang to en. Load Motion Sans before drawing.
Reference source: studio/packs/lp.js

  // @element 10
    CELLS[9] = (k, w, h, u) => {
    frame(k);
    const FX = [ 22, 267 ], FY = 16, FW = 222, FH = 118, ZOOM = .15;
    const cur = t => path([ [ 0, 60, 160 ], [ .5, 120, 70 ], [ 1.8, 140, 84 ], [ 2.2, 370, 70 ], [ 3.4, 390, 84 ], [ 3.9, 470, 160 ], [ 4.8, 60, 160 ] ], t);
    view(k, () => {
      for (let i = 0; i < 2; i++) {
        const inside = t => {
          const [x, y] = cur(t);
          return x > FX[i] && x < FX[i] + FW && y > FY && y < FY + FH;
        };
        const q = trans(inside, u, .5);
        pic(k, FX[i], FY, FW, FH, i, 8, 1 + ZOOM * q);
        txt(k, [ MotionLocale.text("copy_118"), MotionLocale.text("copy_125") ][i], FX[i], FY + FH + 20, `800 13px ${FONT}`, C.ink);
        bars(k, FX[i], FY + FH + 28, 150, 1);
      }
      const [cx, cy] = cur(u);
      cursor(k, cx, cy);
    });
  };
