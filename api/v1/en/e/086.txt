Motion Prompt #086 · Underline animation
Under the link the cursor is on, an orange line grows from the left; when the cursor leaves, the line shrinks away to the right.

- Three links (20 px, bold, 56 apart). The cursor rests on each, left to right, for about 1 s (hit area 6 px wider than the text on each side).
- Underline: orange, 3 thick, 8 px below the baseline.
- On hover, width 0→100% from the left end over 0.3 s (cubic ease-out).
- On leave, the left end moves right over 0.3 s, so the line exits to the right (like scaleX with transform-origin on the right).

Canvas: 539×208; cycle: 4.8s.

  // @element 14
    CELLS[13] = (k, w, h, u) => {
    frame(k);
    const LINKS = [ MotionLocale.text("copy_038"), MotionLocale.text("copy_029"), MotionLocale.text("copy_039") ], F = `800 20px ${FONT}`, GAP = 56, LY = 84, DUR = .3;
    const cur = t => path([ [ 0, 120, 150 ], [ .45, 170, 80 ], [ 1.3, 186, 84 ], [ 1.55, 268, 80 ], [ 2.4, 282, 84 ], [ 2.65, 372, 80 ], [ 3.5, 390, 84 ], [ 3.95, 430, 150 ], [ 4.8, 120, 150 ] ], t);
    view(k, () => {
      logo(k, 22, 28);
      bars(k, 90, 132, 331, 2, C.faint, 6, 12, .6);
      k.font = F;
      const ws = LINKS.map(s => k.measureText(MotionLocale.text(s)).width);
      let x = (VW - ws.reduce((a, b) => a + b, 0) - GAP * (LINKS.length - 1)) / 2;
      LINKS.forEach((s, i) => {
        const x0 = x, x1 = x + ws[i], inside = t => {
          const [a, b] = cur(t);
          return a > x0 - 6 && a < x1 + 6 && b > LY - 22 && b < LY + 14;
        };
        txt(k, s, x0, LY, F, C.ink);
        const on = inside(u);
        let tc = null;
        for (let j = 0, t = u; j < 600; j++, t -= 1 / 120) if (inside(t - 1 / 120) !== on) {
          tc = t;
          break;
        }
        if (tc !== null) {
          const q = eOut(P(u, tc, tc + DUR)), a = on ? 0 : q, b = on ? q : 1;
          if (b > a) {
            k.fillStyle = C.acc;
            k.fillRect(x0 + ws[i] * a, LY + 8, ws[i] * (b - a), 3);
          }
        }
        x += ws[i] + GAP;
      });
      const [cx, cy] = cur(u);
      cursor(k, cx, cy);
    });
  };
