Motion Prompt #070 · Spinner
Clicking "Send" washes the card white and a blue arc spins in the middle; when finished it changes to a check and fades.

- 4.5 s cycle on a 542×191 card. The cursor clicks "Send" (120×40, bottom centre) at 1.0 s.
- Over 1.05–1.2 s (linear) a white scrim (opacity 0.78) and the spinner fade in over the whole card.
- Spinner at (271, 92): a pale ring, radius 26, 6 thick, with a blue 243° (1.35π) arc on top (round caps), turning 1.3 times per second (468°/s).
- Over 2.95–3.05 s the spinner fades; over 2.95–3.2 s a blue circle (radius 28) with a white check scales 0.7 → 1 (easeOutCubic).
- Everything fades out linearly over 3.75–4.0 s.

Canvas: 542×191; cycle: 4.5s.

  // @element 22
    CELLS[21] = ui((w, h, t) => {
    const load = seg(t, 1.05, 1.2) * (1 - seg(t, 2.95, 3.05));
    const done = seg(t, 2.95, 3.15) * (1 - seg(t, 3.75, 4));
    pageLines(24, 18, 494);
    strokeRR(24, 90, 494, 34, 7, C.line2, 1.5);
    fillRR(w / 2 - 60, 140, 120, 40, 8, C.blue);
    text(MotionLocale.text("copy_044"), w / 2, 160, 16, "#fff", 600, "center");
    const ov = Math.max(load, done);
    if (ov > 0) {
      X.fillStyle = `rgba(255,255,255,${.78 * ov})`;
      X.fillRect(0, 0, w, h);
      if (load > 0) {
        X.save();
        X.globalAlpha = load;
        ring(w / 2, 92, 26, "#e3e8f5", 6);
        X.beginPath();
        const a0 = t * Math.PI * 2 * 1.3;
        X.arc(w / 2, 92, 26, a0, a0 + Math.PI * 1.35);
        X.strokeStyle = C.blue;
        X.lineWidth = 6;
        X.lineCap = "round";
        X.stroke();
        X.restore();
      }
      if (done > 0) {
        X.save();
        X.globalAlpha = done;
        const s = .7 + .3 * eo(seg(t, 2.95, 3.2));
        X.translate(w / 2, 92);
        X.scale(s, s);
        circle(0, 0, 28, C.blue);
        check(0, 0, 11, "#fff", 4);
        X.restore();
      }
    }
    pointer([ [ .3, w + 30, h + 30 ], [ .95, w / 2 + 6, 166 ], [ 1.5, w / 2 + 6, 166 ], [ 2.2, w + 30, h + 30 ] ], [ 1 ], t);
  });
