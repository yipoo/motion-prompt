Motion Prompt #157 · Hover image reveal
With the cursor on a row of a text-only list, that row's photo fades in beside the cursor and follows it.

- A 511 × 168 viewport with a text-only list of four rows (39 px rows, 17 px names). The cursor moves down the rows, about 1 s on each. One pass takes 4.7 s (twice per 9.4 s pack cycle).
- Hovered row: over 0.25 s (cubic ease-out) the name shifts 8 px right and the name and the arrow at the right turn orange; on leave they return over 0.25 s.
- Photo: that row's photo (132 × 84, radius 6, drop shadow) appears 26 px right of the cursor, a little above it, kept inside the window vertically.
- Over the same 0.25 s the photo goes from opacity 0 to 1 and from 0.85× to 1×. It follows the cursor without lag; moving to the next row fades the old photo out as the new one comes in.

Canvas: 539×208; cycle: 9.4s.

  // @element 13
    CELLS[12] = (k, w, h, u) => {
    u = snapU(u);
    frame(k);
    const NAMES = [ MotionLocale.text("copy_124"), MotionLocale.text("copy_121"), MotionLocale.text("copy_120"), MotionLocale.text("copy_122") ], RT = 6, RH = 39;
    const cur = t => path([ [ 0, 140, 200 ], [ .4, 150, 26 ], [ 1.15, 190, 28 ], [ 1.4, 205, 65 ], [ 2.15, 230, 67 ], [ 2.4, 245, 104 ], [ 3.15, 265, 106 ], [ 3.4, 280, 143 ], [ 3.9, 292, 144 ], [ 4.3, 150, 200 ], [ 4.7, 140, 200 ] ], t);
    const [cx, cy] = cur(u);
    view(k, () => {
      NAMES.forEach((s, i) => {
        const top = RT + i * RH, inside = t => {
          const [a, b] = cur(t);
          return b > top && b < top + RH && a > 20 && a < 490;
        };
        const q = trans(inside, u, .25);
        txt(k, "0" + (i + 1), 30, top + 25, `800 12px ${EN}`, C.sub);
        txt(k, s, 64 + 8 * q, top + 26, `800 17px ${FONT}`, mix(C.ink, C.acc, q));
        txt(k, "→", 476, top + 25, `700 15px ${EN}`, mix("#B7BFCC", C.acc, q), "right");
        k.fillStyle = C.rule;
        k.fillRect(24, top + RH, VW - 48, 1);
      });
      NAMES.forEach((s, i) => {
        const top = RT + i * RH, q = trans(t => {
          const [a, b] = cur(t);
          return b > top && b < top + RH && a > 20 && a < 490;
        }, u, .25);
        if (q <= 0) return;
        const iw = 132, ih = 84, px = cx + 26, py = clamp(cy - 46, 4, VH - ih - 4), sc = .85 + .15 * q;
        k.save();
        k.globalAlpha = q;
        k.translate(px + iw / 2, py + ih / 2);
        k.scale(sc, sc);
        k.shadowColor = "rgba(28,35,51,.25)";
        k.shadowBlur = 12;
        k.shadowOffsetY = 4;
        k.fillStyle = "#fff";
        rr(k, -iw / 2, -ih / 2, iw, ih, 6);
        k.fill();
        k.shadowColor = "transparent";
        pic(k, -iw / 2, -ih / 2, iw, ih, [ 0, 2, 1, 2 ][i], 6);
        k.restore();
      });
      cursor(k, cx, cy);
    });
  };
