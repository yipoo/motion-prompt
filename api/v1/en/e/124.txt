Motion Prompt #124 · Wiggle
A bell keeps jittering irregularly: not one hit, but a small shake that never stops.

- A yellow bell pivoting at its top (centre, y = 34). It never stops moving.
- Angle = the sum of four sine waves: 23, 31, 43 and 59 cycles per 4.7 s (about 4.9, 6.6, 9.1 and 12.6 Hz), amplitudes 0.085, 0.065, 0.05 and 0.035 rad (up to about 0.235 rad, 13.5°, when they line up).
- Position also jitters: x with 29, 47 and 61 cycles (4, 3, 2 px), y with 37 and 53 cycles (2.4, 1.8 px).
- Every frequency is a whole number of cycles per 4.7 s, so the loop joins up; the counts share no common factor, so it does not look regular.
- The clapper (coral, radius 13) swings the other way (angle × 70 px). White vibration arcs (radius 86 and 110, width 6) flicker on both sides the whole time.

Canvas: 539×208; cycle: 4.7s.

  // @element 04
    const WGF = 2 * Math.PI / CYC;
  const wsum = (u, arr) => arr.reduce((s, [kk, am, ph]) => s + am * Math.sin(WGF * kk * u + ph), 0);
  CELLS[3] = (k, w, h, u) => {
    stage(k);
    const rot = wsum(u, [ [ 23, .085, .3 ], [ 31, .065, 1.9 ], [ 43, .05, 4.2 ], [ 59, .035, 2.6 ] ]);
    const dx = wsum(u, [ [ 29, 4, .7 ], [ 47, 3, 3.1 ], [ 61, 2, 5 ] ]), dy = wsum(u, [ [ 37, 2.4, 2.2 ], [ 53, 1.8, .4 ] ]);
    const X = w / 2 + dx, Y = 34 + dy;
    k.strokeStyle = "#fff";
    k.lineCap = "round";
    k.lineWidth = 6;
    [ [ 86, .85 ], [ 110, .5 ] ].forEach(([r, al], j) => {
      const fl = .55 + .45 * Math.abs(Math.sin(WGF * (41 + j * 12) * u));
      k.globalAlpha = al * fl;
      [ 0, Math.PI ].forEach(base => {
        k.beginPath();
        k.arc(w / 2, 104, r, base - .42, base + .42);
        k.stroke();
      });
    });
    k.globalAlpha = 1;
    k.save();
    k.translate(X, Y);
    k.rotate(rot);
    k.fillStyle = C.coral;
    k.beginPath();
    k.arc(-rot * 70, 128, 13, 0, 7);
    k.fill();
    k.fillStyle = C.yel;
    k.beginPath();
    k.arc(0, 8, 10, 0, 7);
    k.fill();
    k.beginPath();
    k.moveTo(-50, 112);
    k.quadraticCurveTo(-42, 100, -42, 76);
    k.bezierCurveTo(-42, 36, -22, 18, 0, 18);
    k.bezierCurveTo(22, 18, 42, 36, 42, 76);
    k.quadraticCurveTo(42, 100, 50, 112);
    k.closePath();
    k.fill();
    rr(k, -60, 106, 120, 16, 8);
    k.fill();
    k.fillStyle = "rgba(255,255,255,.35)";
    rr(k, -28, 40, 12, 50, 6);
    k.fill();
    k.restore();
  };
