Motion Prompt #095 · Animated gradient
Behind the headline, soft patches of orange, light orange, blue and light blue drift in circles and keep blending differently.

- Base colour #FFF1EB, with four radial gradients on top (centre colour → transparent at the edge, each at 0.85 opacity).
- Orange (radius 230), light orange #FFB08F (210), blue (240), light blue #9DB6FB (200).
- Each centre travels an ellipse once per 4.8 s (x radii 70, 60, 80, 50; y radii 30, 40, 30, 35; start angles 0, 2.1, 4.2, 1.0 rad).
- A white headline (34 px) and a button sit on top; only the gradient moves.

Canvas: 539×208; cycle: 4.8s.

  // @element 23
    CELLS[22] = (k, w, h, u) => {
    frame(k);
    const a = 2 * Math.PI * u / CYC;
    const BLOBS = [ [ C.acc, 150, 60, 230, 70, 30, 0 ], [ "#FFB08F", 380, 40, 210, 60, 40, 2.1 ], [ C.blue, 330, 150, 240, 80, 30, 4.2 ], [ "#9DB6FB", 90, 170, 200, 50, 35, 1 ] ];
    view(k, () => {
      k.fillStyle = "#FFF1EB";
      k.fillRect(0, 0, VW, VH);
      for (const [col, x0, y0, r, ax, ay, ph] of BLOBS) {
        const x = x0 + ax * Math.cos(a + ph), y = y0 + ay * Math.sin(a + ph);
        const g = k.createRadialGradient(x, y, 0, x, y, r);
        g.addColorStop(0, col);
        g.addColorStop(1, "rgba(255,255,255,0)");
        k.globalAlpha = .85;
        k.fillStyle = g;
        k.fillRect(0, 0, VW, VH);
        k.globalAlpha = 1;
      }
      txt(k, MotionLocale.text("Hello, Future."), 40, 82, `800 34px ${EN}`, "#fff", "left", "alphabetic", "1px");
      bars(k, 40, 100, 200, 2, "rgba(255,255,255,.8)", 6, 12, .6);
      btn(k, 40, 128, 104, 26, MotionLocale.text("copy_099"), "#fff", C.ink);
    });
  };
