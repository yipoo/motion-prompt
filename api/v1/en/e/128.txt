Motion Prompt #128 · Stop-motion style
A star hops across in choppy steps at about 8 frames per second, each frame slightly off.

- 38 frames per 4.7 s (about 8 per second, each held for about 0.124 s). Only those frames are drawn; nothing is drawn in between.
- A yellow star (radius 38) hops from x = −44 to 583 along three arches of height 54·|sin(3πt)|, turning twice.
- Each frame gets a fixed random jitter: position ±5 px across and ±4.5 px up and down, rotation ±0.15 rad, size ±5%.
- The previous two frames stay faintly visible (opacity 0.28 and 0.12), like onion skin.
- Black film-strip bands top and bottom (26 px each, with sprocket holes) and a frames-per-second label.

Canvas: 539×208; cycle: 4.7s.

  // @element 08
    CELLS[7] = (k, w, h, u) => {
    u = snapU(u);
    stage(k);
    k.fillStyle = "#0A1230";
    k.fillRect(0, 0, w, 26);
    k.fillRect(0, h - 26, w, 26);
    k.fillStyle = C.navy3;
    for (let x = 12; x < w; x += 34) {
      rr(k, x, 8, 18, 11, 3);
      k.fill();
      rr(k, x, h - 19, 18, 11, 3);
      k.fill();
    }
    const NS = 38, st = CYC / NS, fi = Math.floor(u / st), floor = h - 44;
    const pose = f => {
      const t = f / NS, j = i => H01(f * 7919 + i * 131 + 77) - .5;
      return {
        x: lerp(-44, w + 44, t) + j(1) * 10,
        y: floor - 36 - 54 * Math.abs(Math.sin(t * Math.PI * 3)) + j(2) * 9,
        r: t * Math.PI * 4 + j(3) * .3,
        s: 1 + j(4) * .1
      };
    };
    const draw = (p, a) => {
      k.globalAlpha = a;
      k.save();
      k.translate(p.x + 5, p.y + 6);
      k.rotate(p.r);
      k.scale(p.s, p.s);
      k.fillStyle = "rgba(0,0,0,.35)";
      star(k, 0, 0, 38, 0, .5);
      k.fill();
      k.restore();
      k.save();
      k.translate(p.x, p.y);
      k.rotate(p.r);
      k.scale(p.s, p.s);
      k.fillStyle = C.yel;
      star(k, 0, 0, 38, 0, .5);
      k.fill();
      k.fillStyle = C.coral;
      k.beginPath();
      k.arc(0, 2, 9, 0, 7);
      k.fill();
      k.restore();
      k.globalAlpha = 1;
    };
    draw(pose(mod(fi - 2, NS)), .12);
    draw(pose(mod(fi - 1, NS)), .28);
    const p = pose(fi);
    k.fillStyle = "rgba(0,0,0,.3)";
    k.beginPath();
    k.ellipse(p.x, floor + 6, 30, 5, 0, 0, 7);
    k.fill();
    draw(p, 1);
    txt(k, MotionLocale.text("copy_014"), w - 16, 50, `800 24px ${FONT}`, C.dim, "right", "middle");
  };
