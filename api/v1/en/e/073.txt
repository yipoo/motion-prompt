Motion Prompt #073 · Parallax scrolling
As the page scrolls down, the headline moves up normally while the sky and sun, far hills and near hills move more slowly.

- A browser window with a 511 × 168 viewport. Scroll: 0→120 px at 0.3–1.9 s, 120→0 at 2.9–4.3 s (cubic ease-in-out).
- Each layer moves by scroll × factor: sky and sun 0.15, far hills 0.4, near hills 0.7.
- Headline (28 px), button and the next section move at 1.0 (normal scrolling).
- The next section (white) starts at the bottom of the hero (168 px) and slides over the slower background.
- The scrollbar thumb on the right moves with the scroll.

Canvas: 539×208; cycle: 4.8s.

  // @element 01
    CELLS[0] = (k, w, h, u) => {
    frame(k);
    const s = steps([ [ .3, 0 ], [ 1.9, 1 ], [ 2.9, 1 ], [ 4.3, 0 ] ], u), y = s * 120;
    view(k, () => {
      const g = k.createLinearGradient(0, 0, 0, 200);
      g.addColorStop(0, "#9DB6FB");
      g.addColorStop(1, "#E6ECFF");
      k.save();
      k.translate(0, -y * .15);
      k.fillStyle = g;
      k.fillRect(0, 0, VW, 260);
      k.fillStyle = C.acc;
      k.beginPath();
      k.arc(392, 54, 25, 0, 7);
      k.fill();
      k.restore();
      k.save();
      k.translate(0, -y * .4);
      mount(k, VW, 190, M1, "#7D9AF4", 0);
      k.restore();
      k.save();
      k.translate(0, -y * .7);
      mount(k, VW, 205, M2, "#4B6BDB", 1);
      k.restore();
      k.save();
      k.translate(0, -y);
      txt(k, MotionLocale.text("copy_111"), 34, 64, `800 28px ${FONT}`, "#fff", "left", "alphabetic", "1px");
      bars(k, 34, 78, 160, 2, "rgba(255,255,255,.85)");
      btn(k, 34, 106, 92, 26, MotionLocale.text("copy_134"));
      k.fillStyle = "#fff";
      k.fillRect(0, VH, VW, 200);
      txt(k, MotionLocale.text("copy_128"), 34, VH + 34, `800 16px ${FONT}`, C.ink);
      for (let i = 0; i < 3; i++) {
        pic(k, 34 + i * 152, VH + 50, 136, 58, i, 6);
        bars(k, 34 + i * 152, VH + 116, 110, 1, C.faint);
      }
      k.restore();
      sbar(k, s * .35, .4);
    });
