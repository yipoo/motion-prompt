Motion Prompt #125 · Pulse
A heart throbs like a heartbeat, a double beat each time, with rings spreading out.

- A coral heart. One beat per 1.175 s (four per 4.7 s); each beat is a double "lub-dub".
- Scale = 1 + 0.26·G(0.2 s, width 0.065) + 0.15·G(0.44 s, width 0.07), where G = e^(−((t − centre) ÷ width)²): a quick swell that drops straight back.
- Two heart-shaped outlines spread from 0.16 s and 0.40 s: scale 1.2→2.4 over 0.8 s, cubic ease-out.
- As they spread, their opacity goes 0.75→0 (the second at 60%) and the line width 8.5→1.5 px.

Canvas: 539×208; cycle: 4.7s.

  // @element 05
    CELLS[4] = (k, w, h, u) => {
    stage(k);
    const T = CYC / 4, lu = mod(u, T), X = w / 2, Y = h / 2 + 10;
    const g = (c, s) => Math.exp(-Math.pow((lu - c) / s, 2));
    const b = .26 * g(.2, .065) + .15 * g(.44, .07) + .26 * Math.exp(-Math.pow((lu - T - .2) / .065, 2));
    for (const [t0, s] of [ [ .16, 1 ], [ .4, .6 ] ]) {
      const q = P(lu, t0, t0 + .8);
      if (q <= 0 || q >= 1) continue;
      k.save();
      k.translate(X, Y);
      const sc = 1.2 + 1.2 * eOut(q);
      k.scale(sc, sc);
      heartPath(k, 44);
      k.strokeStyle = C.coral;
      k.globalAlpha = (1 - q) * .75 * s;
      k.lineWidth = (7 * (1 - q) + 1.5) / sc;
      k.stroke();
      k.restore();
    }
    k.globalAlpha = 1;
    k.save();
    k.translate(X, Y);
    k.scale(1 + b, 1 + b);
    heartPath(k, 44);
    k.fillStyle = C.coral;
    k.fill();
    k.fillStyle = "rgba(255,255,255,.75)";
    k.beginPath();
    k.ellipse(-30, -28, 9, 6, -.7, 0, 7);
    k.fill();
    k.restore();
  };
