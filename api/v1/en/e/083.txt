Motion Prompt #083 · Cursor follower
As the cursor moves around the page, an orange circle follows it a little behind.

- Cursor path: x = 255 + 175·sin(2πt/4.8), y = 86 + 46·sin(4πt/4.8) (a figure eight lying on its side, one lap per 4.8 s).
- Circle: radius 17, orange, opacity 0.88, drawn under the cursor.
- Follow rule: every frame (60 fps) it covers 12% of the remaining distance (x += (cursor x − x) × 0.12).
- The arrow cursor stays visible.

Canvas: 539×208; cycle: 4.8s.

  // @element 11
    CELLS[10] = (k, w, h, u) => {
    frame(k);
    const cur = t => [ 255 + 175 * Math.sin(2 * Math.PI * t / CYC), 86 + 46 * Math.sin(4 * Math.PI * t / CYC) ];
    let t = u - 2, [fx, fy] = cur(t);
    for (let i = 0; i < 120; i++) {
      t += 1 / 60;
      const [tx, ty] = cur(t);
      fx += (tx - fx) * .12;
      fy += (ty - fy) * .12;
    }
    view(k, () => {
      txt(k, MotionLocale.text("CREATIVE STUDIO"), VW / 2, 74, `800 34px ${EN}`, C.ink, "center", "alphabetic", "3px");
      bars(k, VW / 2 - 120, 96, 240, 2, C.faint, 6, 12, .7);
      btn(k, VW / 2 - 50, 128, 100, 24, MotionLocale.text("copy_135"), C.ink);
      k.fillStyle = C.acc;
      k.globalAlpha = .88;
      k.beginPath();
      k.arc(fx, fy, 17, 0, 7);
      k.fill();
      k.globalAlpha = 1;
      const [cx, cy] = cur(u);
      cursor(k, cx, cy);
    });
  };
