Motion Prompt #189 · Cutaway
From the speaker to the listening audience and back; what is cut in is a different subject.

- Join: two straight cuts (0 frames); to the other subject at 1.7 s and back to the scene at 3.0 s (1.3 s away)
- Main shot (A): fixed camera (height 1.4 m); a person 5.5 m away talks, swinging arms and legs by 0.45·sin(u×6) rad, with a grey lectern 0.9 m to the right
- Other subject (CUTAWAY): the listening audience, five people 3.1–4.6 m away in blue and pale blue with both arms up, bouncing by |sin(t×9+i)|×5 cm; no background (a different place)
- Bottom strip: A (grey), other subject (blue), A (grey) with a playhead (17 Insert shot shows a detail of the same scene)
- Top-right readout shows "SHOT A" → "CUTAWAY" → "SHOT A"; at 4.7 s it cuts straight back to the start

Canvas: 539×208; cycle: 9.4s.

  // @element 21
    const CUT = [ 1.7, 3 ];
  CELLS[20] = shot((k, w, h, u) => {
    const away = u >= CUT[0] && u < CUT[1];
    if (!away) world(k, w, h, cam({
      y: 1.4
    }), {
      person: {
        x: 0,
        z: 5.5,
        walk: .45 * Math.sin(u * 6)
      },
      extra: [ [ .9, 0, 5.6, kk => {
        kk.fillStyle = C.g3;
        kk.fillRect(-.3, 0, .6, 1.1);
        kk.fillRect(-.4, 1.05, .8, .1);
      } ] ]
    }); else {
      const t = u - CUT[0], ex = [];
      [ [ -1.6, 3.2 ], [ -.2, 3.6 ], [ 1.3, 3.1 ], [ .6, 4.4 ], [ -1.1, 4.6 ] ].forEach(([x, z], i) => ex.push([ x, 0, z, kk => {
        kk.save();
        kk.translate(0, Math.abs(Math.sin(t * 9 + i)) * .05);
        person(kk, 1, {
          armUp: true,
          col: i % 2 ? C.blue : C.blueM
        });
        kk.restore();
      } ]));
      world(k, w, h, cam({
        y: 1.6,
        pitch: -.08
      }), {
        buildings: false,
        trees: false,
        mountains: false,
        sun: null,
        extra: ex
      });
    }
    strip(k, w, h, [ [ 0, CUT[0], "A", C.g2 ], [ CUT[0], CUT[1], MotionLocale.text("copy_056"), C.blue ], [ CUT[1], CYC, "A", C.g2 ] ], u);
    return away ? MotionLocale.text("CUTAWAY") : MotionLocale.text("SHOT A");
  });
