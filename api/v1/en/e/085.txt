Motion Prompt #085 · Ripple effect
Each click sends a white circle out from the click point to fill the button, then it fades away.

- Button 280 × 56 (orange, radius 10). Clicks at 0.7 s (left of centre) and 2.9 s (right of centre).
- The circle is centred on the click point; its radius grows from 0 to the distance to the button's farthest corner over 0.6 s (cubic ease-out).
- White at 0.4 opacity, fading to 0 between 0.3 and 0.8 s after the click.
- The circle is clipped to the button (overflow: hidden); the label sits above it.

Canvas: 539×208; cycle: 4.8s.

  // @element 13
    CELLS[12] = (k, w, h, u) => {
    frame(k);
    const B = {
      x: 115,
      y: 78,
      w: 280,
      h: 56
    }, TAPS = [ [ .7, 165, 110 ], [ 2.9, 345, 100 ] ], DUR = .6;
    const [cx, cy] = path([ [ 0, 430, 160 ], [ .55, 165, 110 ], [ 1.6, 165, 110 ], [ 2.7, 345, 100 ], [ 3.6, 345, 100 ], [ 4.4, 430, 160 ], [ 4.8, 430, 160 ] ], u);
    view(k, () => {
      k.fillStyle = C.soft;
      k.fillRect(0, 0, VW, VH);
      txt(k, MotionLocale.text("copy_131"), VW / 2, 38, `800 18px ${FONT}`, C.ink, "center");
      bars(k, VW / 2 - 110, 50, 220, 1, C.faint, 6, 11, 1);
      k.save();
      k.shadowColor = "rgba(28,35,51,.18)";
      k.shadowBlur = 8;
      k.shadowOffsetY = 3;
      k.fillStyle = C.acc;
      rr(k, B.x, B.y, B.w, B.h, 10);
      k.fill();
      k.restore();
      k.save();
      rr(k, B.x, B.y, B.w, B.h, 10);
      k.clip();
      for (const [tc, px, py] of TAPS) {
        const d = u - tc;
        if (d < 0 || d > .8) continue;
        const Rm = Math.max(Math.hypot(px - B.x, py - B.y), Math.hypot(px - B.x - B.w, py - B.y), Math.hypot(px - B.x, py - B.y - B.h), Math.hypot(px - B.x - B.w, py - B.y - B.h));
        k.fillStyle = `rgba(255,255,255,${.4 * (1 - P(d, .3, .8))})`;
        k.beginPath();
        k.arc(px, py, Rm * eOut(P(d, 0, DUR)), 0, 7);
        k.fill();
      }
      k.restore();
      txt(k, MotionLocale.text("copy_137"), VW / 2, B.y + B.h / 2 + 1, `800 17px ${FONT}`, "#fff", "center", "middle");
      click(k, cx, cy, u, TAPS.map(a => a[0]));
    });
  };
