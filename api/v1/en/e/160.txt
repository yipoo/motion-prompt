Motion Prompt #160 · Loading button
Clicking the submit button shrinks it to a circle with a spinner inside; when it finishes it widens into a green "Done" with a check.

- A 511 × 168 viewport with an orange pill button, 220 × 36, labeled "Send" (Send). It is clicked at 0.8 s. One pass takes 4.7 s (twice per 9.4 s pack cycle).
- Over 0.85–1.15 s (0.3 s, cubic ease-in-out) it shrinks from 220 to 36 px wide, a circle; the label fades in the first third of the shrink.
- From 1.1 to 2.3 s a white spinner (radius 10, line width 3, 0.7π arc) turns once every 0.7 s inside the circle.
- Over 2.25–2.45 s the color goes orange → green (#1E9E5A) and a white check draws over 2.25–2.5 s; over 2.45–2.8 s (cubic ease-in-out) it widens 36 → 150 px, the check moves 26 px left and "完了" (Done) appears.
- It fades out over 3.9–4.1 s and the original "Send" button comes back, rising 6 px, over 4.1–4.4 s (cubic ease-out).

Canvas: 539×208; cycle: 9.4s.

  // @element 16
    CELLS[15] = (k, w, h, u) => {
    u = snapU(u);
    frame(k);
    const CLK = .8, BY = 124, BHT = 36;
    const shr = eIO(P(u, CLK + .05, CLK + .35)), done = eOut(P(u, 2.25, 2.45)), grow = eIO(P(u, 2.45, 2.8)), fade = P(u, 3.9, 4.1), back = eOut(P(u, 4.1, 4.4));
    const bw = fade >= 1 ? 220 : grow > 0 ? lerp(BHT, 150, grow) : lerp(220, BHT, shr);
    const cur = t => path([ [ 0, 430, 205 ], [ .55, 290, 136 ], [ 1.1, 290, 136 ], [ 1.6, 440, 205 ], [ 4.7, 430, 205 ] ], t);
    const [cx, cy] = cur(u);
    view(k, () => {
      k.fillStyle = C.blueL;
      k.fillRect(0, 0, VW, VH);
      k.save();
      k.shadowColor = "rgba(28,35,51,.1)";
      k.shadowBlur = 12;
      k.shadowOffsetY = 3;
      k.fillStyle = "#fff";
      rr(k, 130, 10, 251, 150, 12);
      k.fill();
      k.restore();
      txt(k, MotionLocale.text("copy_052"), VW / 2, 36, `800 15px ${FONT}`, C.ink, "center");
      for (let i = 0; i < 2; i++) {
        k.strokeStyle = "#D5DBE6";
        k.lineWidth = 1.5;
        rr(k, 150, 50 + i * 32, 211, 24, 6);
        k.stroke();
        bars(k, 160, 59 + i * 32, i ? 90 : 120, 1, "#AEB6C5", 6);
      }
      const col = fade >= 1 ? "#FF5A2C" : mix("#FF5A2C", GREEN, done), ba = fade >= 1 ? back : 1 - fade;
      k.save();
      k.globalAlpha = ba;
      k.translate(0, fade >= 1 ? 6 * (1 - back) : 0);
      k.save();
      k.shadowColor = "rgba(28,35,51,.18)";
      k.shadowBlur = 8;
      k.shadowOffsetY = 3;
      k.fillStyle = col;
      rr(k, VW / 2 - bw / 2, BY, bw, BHT, BHT / 2);
      k.fill();
      k.restore();
      const idleA = fade >= 1 ? 1 : 1 - clamp(shr * 3);
      if (idleA > 0) {
        k.save();
        k.globalAlpha = idleA * ba;
        txt(k, MotionLocale.text("copy_137"), VW / 2, BY + BHT / 2 + .5, `800 14px ${FONT}`, "#fff", "center", "middle");
        k.restore();
      }
      if (u > CLK + .3 && u < 2.3) {
        k.save();
        k.globalAlpha = clamp((u - CLK - .3) / .1) * (1 - clamp((u - 2.2) / .08));
        k.lineCap = "round";
        k.lineWidth = 3;
        k.strokeStyle = "rgba(255,255,255,.35)";
        k.beginPath();
        k.arc(VW / 2, BY + BHT / 2, 10, 0, 7);
        k.stroke();
        const a = u / .7 * Math.PI * 2;
        k.strokeStyle = "#fff";
        k.beginPath();
        k.arc(VW / 2, BY + BHT / 2, 10, a, a + Math.PI * .7);
        k.stroke();
        k.restore();
      }
      if (u >= 2.25 && fade < 1) {
        k.save();
        const ox = lerp(0, -26, grow);
        check(k, VW / 2 + ox, BY + BHT / 2, 16, "#fff", 3, P(u, 2.25, 2.5));
        if (grow > .5) {
          k.globalAlpha = ba * P(grow, .5, 1);
          txt(k, MotionLocale.text("copy_042"), VW / 2 + 12, BY + BHT / 2 + .5, `800 14px ${FONT}`, "#fff", "center", "middle");
        }
        k.restore();
      }
      k.restore();
      click(k, cx, cy, u, [ CLK ]);
    });
  };
