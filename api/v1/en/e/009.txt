Motion Prompt #009 · Easing
Two squares cover the same distance in the same time: the top one at constant speed, the bottom one eased.

- One round trip per 4.7 s. Two 62 px squares (radius 16) travel 316 px horizontally: out at 0.2–1.75 s (1.55 s), hold 0.8 s, back at 2.55–4.1 s.
- Top (constant): linear motion; shape and colour never change.
- Bottom (eased): cubic-bezier(.86, 0, .14, 1). Slow start, very fast middle, slow stop.
- The bottom square stretches horizontally up to 1.5× and flattens to 0.8× in proportion to speed (speed ÷ 3.6, clamped to 0–1).
- Behind it: 3 speed lines (up to 150 px × 0.8–1.25) and 4 ghost copies lagging 0.035 s each.
- Each rail shows 12 dots marking positions at equal time intervals (evenly spaced on top, bunched at both ends below).

Canvas: 539×208; cycle: 4.7s.

    const EZ = bez(.86, 0, .14, 1);
  const EZT = [ .2, 1.75, 2.55, 4.1 ];
  const ezPos = u => {
    if (u < EZT[0]) return 0;
    if (u < EZT[1]) return P(u, EZT[0], EZT[1]);
    if (u < EZT[2]) return 1;
    if (u < EZT[3]) return 1 - P(u, EZT[2], EZT[3]);
    return 0;
  };
  const ezE = u => {
    if (u < EZT[0]) return 0;
    if (u < EZT[1]) return EZ(P(u, EZT[0], EZT[1]));
    if (u < EZT[2]) return 1;
    if (u < EZT[3]) return 1 - EZ(P(u, EZT[2], EZT[3]));
    return 0;
  };
  CELLS[8] = (k, w, h, u) => {
    stage(k);
    const x0 = 162, x1 = 478, D = x1 - x0, y1 = 58, y2 = 150, S = 62;
    const lin = ezPos(u), mv = u >= EZT[0] && u < EZT[1] ? 1 : u >= EZT[2] && u < EZT[3] ? -1 : 0, pp = mv === 1 ? P(u, EZT[0], EZT[1]) : mv === -1 ? P(u, EZT[2], EZT[3]) : 0;
    const ez = mv === 1 ? EZ(pp) : mv === -1 ? 1 - EZ(pp) : lin;
    const vel = mv ? (EZ(Math.min(1, pp + .01)) - EZ(Math.max(0, pp - .01))) / .02 : 0, vn = clamp(vel / 3.6);
    txt(k, MotionLocale.text("copy_008"), 24, y1 + 1, `700 34px ${FONT}`, C.dim, "left", "middle");
    txt(k, MotionLocale.text("copy_009"), 24, y2 + 1, `700 34px ${FONT}`, C.yel, "left", "middle");
    [ [ y1, x => x, "#34489A" ], [ y2, EZ, "#34489A" ] ].forEach(([y, f, col]) => {
      k.strokeStyle = C.navy2;
      k.lineWidth = 6;
      k.lineCap = "round";
      k.beginPath();
      k.moveTo(x0, y);
      k.lineTo(x1, y);
      k.stroke();
      k.fillStyle = col;
      for (let i = 0; i <= 11; i++) {
        k.beginPath();
        k.arc(x0 + f(i / 11) * D, y, 5, 0, 7);
        k.fill();
      }
    });
    k.fillStyle = "#6677AD";
    rr(k, x0 + lin * D - S / 2, y1 - S / 2, S, S, 16);
    k.fill();
    const X = x0 + ez * D;
    if (vn > .05) {
      k.lineCap = "round";
      [ [ -20, C.yel, 1 ], [ 0, C.coral, 1.25 ], [ 20, C.yel, .8 ] ].forEach(([dy, col, m]) => {
        k.strokeStyle = col;
        k.globalAlpha = .9 * vn;
        k.lineWidth = 10;
        k.beginPath();
        k.moveTo(X - mv * (S / 2), y2 + dy);
        k.lineTo(X - mv * (S / 2 + 150 * vn * m), y2 + dy);
        k.stroke();
      });
      k.globalAlpha = 1;
    }
    for (let g = 4; g >= 1; g--) {
      const gx = x0 + ezE(u - g * .035) * D;
      if (Math.abs(gx - X) > 3) {
        k.globalAlpha = .16 * (5 - g) / 4 + .06;
        k.fillStyle = C.coral;
        rr(k, gx - S / 2, y2 - S / 2, S, S, 16);
        k.fill();
      }
    }
    k.globalAlpha = 1;
    const sx = 1 + .5 * vn, sy = 1 - .2 * vn;
    k.save();
    k.translate(X, y2);
    k.scale(sx, sy);
    k.fillStyle = C.coral;
    rr(k, -S / 2, -S / 2, S, S, 16);
    k.fill();
    k.fillStyle = "rgba(255,255,255,.9)";
    k.beginPath();
    k.arc(S / 2 - 18, -S / 2 + 18, 6, 0, 7);
    k.fill();
    k.restore();
  };
