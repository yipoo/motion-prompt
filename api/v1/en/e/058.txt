Motion Prompt #058 · Carousel
Each click on the right arrow slides a row of three different images one step to the left, and the dot below moves.

- 4.5 s cycle on a 542×191 card. Slides 330×138, radius 8, centred with an 18 px gap; the neighbours peek in at the edges.
- White round buttons (radius 18, shadow blur 10) with chevrons at left and right; three dots below (current one blue 16×8, others 8×8).
- The cursor clicks the right arrow (514, 88) at 1.0, 2.1 and 3.2 s.
- Each click moves the row 348 px left (one slide plus gap) over 0.4 s with easeInOutCubic.
- The three pictures (landscape, three circles, bar chart) repeat, so after three moves the first is back.

Canvas: 542×191; cycle: 4.5s.

  // @element 10
    CELLS[9] = ui((w, h, t) => {
    const p = eio(seg(t, 1, 1.4)) + eio(seg(t, 2.1, 2.5)) + eio(seg(t, 3.2, 3.6));
    const sw = 330, gap = 18, cx = w / 2;
    X.save();
    X.beginPath();
    X.rect(0, 0, w, 160);
    X.clip();
    for (let j = Math.floor(p) - 2; j <= Math.floor(p) + 2; j++) {
      const x = cx - sw / 2 + (j - p) * (sw + gap), idx = (j % 3 + 3) % 3;
      if (idx === 0) mountains(x, 14, sw, 138);
      if (idx === 1) {
        fillRR(x, 14, sw, 138, 8, "#e3ebfd");
        circle(x + 90, 83, 34, C.mb);
        circle(x + 165, 83, 34, C.sb);
        circle(x + 240, 83, 34, C.mb);
      }
      if (idx === 2) {
        fillRR(x, 14, sw, 138, 8, C.lb);
        for (let k = 0; k < 5; k++) {
          const bh = [ 40, 70, 55, 95, 80 ][k];
          fillRR(x + 60 + k * 46, 132 - bh, 30, bh, 4, k === 3 ? C.blue : C.mb);
        }
      }
    }
    X.restore();
    [ [ 30, 83, Math.PI / 2 ], [ w - 30, 83, -Math.PI / 2 ] ].forEach(([x, y, r]) => {
      shadowed(() => circle(x, y, 18, "#fff"), 10, 3, .2);
      chevron(x, y, 5, r, "#555");
    });
    const act = Math.round(p) % 3;
    for (let k = 0; k < 3; k++) {
      const on = k === act;
      fillRR(w / 2 - 30 + k * 22 - (on ? 4 : 0), 172, on ? 16 : 8, 8, 4, on ? C.blue : C.line2);
    }
    pointer([ [ .3, w + 30, h + 30 ], [ .95, w - 28, 88 ], [ 3.4, w - 28, 88 ], [ 4.2, w + 30, h + 30 ] ], [ 1, 2.1, 3.2 ], t);
  });
