Motion Prompt #015 · Particles
Particles keep shooting up from an emitter, spread out and fall.

- 84 particles, emitted from the bottom centre (horizontal centre, 20 px above the bottom edge).
- Initial velocity: ±190 px/s horizontally, 250–400 px/s upward. Gravity 470 px/s² (parabolic paths).
- Each particle repeats with a period of 2.35 s or about 1.57 s, with a random start offset (fixed seed, so every run looks the same).
- Lifetime up to 1.5 s: fade in over the first 0.06 s, fade out over the last 30%. Scale 1.15→0.7.
- Shapes are circles, squares and triangles (size 5–11 px) in blue, coral, yellow and mint, spinning at up to ±4.5 rad/s.

Canvas: 539×208; cycle: 4.7s.

  // @element 15
    const PARTS = (() => {
    const R = rand(15), a = [];
    for (let i = 0; i < 84; i++) {
      const n = R() < .5 ? 2 : 3;
      a.push({
        T: CYC / n,
        ph: R() * CYC,
        vx: (R() - .5) * 2 * 190,
        vy: -(250 + R() * 150),
        sz: 5 + R() * 6,
        col: PAL[i % 4],
        kind: i % 3,
        spin: (R() - .5) * 9
      });
    }
    return a;
  })();
  CELLS[14] = (k, w, h, u) => {
    stage(k);
    const X = w / 2, Y = h - 20, G = 470;
    for (const p of PARTS) {
      const t = mod(u + p.ph, p.T), life = Math.min(p.T, 1.5);
      if (t > life) continue;
      const x = X + p.vx * t, y = Y + p.vy * t + .5 * G * t * t, a = Math.min(1, t / .06) * (1 - smooth(P(t, life * .7, life))), sc = lerp(1.15, .7, t / life);
      k.globalAlpha = a;
      k.fillStyle = p.col;
      k.save();
      k.translate(x, y);
      k.rotate(t * p.spin);
      k.scale(sc, sc);
      if (p.kind === 0) {
        k.beginPath();
        k.arc(0, 0, p.sz, 0, 7);
        k.fill();
      } else if (p.kind === 1) {
        k.fillRect(-p.sz, -p.sz, p.sz * 2, p.sz * 2);
      } else {
        k.beginPath();
        k.moveTo(0, -p.sz * 1.3);
        k.lineTo(p.sz * 1.2, p.sz);
        k.lineTo(-p.sz * 1.2, p.sz);
        k.closePath();
        k.fill();
      }
      k.restore();
    }
    k.globalAlpha = 1;
    k.fillStyle = C.navy3;
    rr(k, X - 70, h - 12, 140, 30, 10);
    k.fill();
    k.fillStyle = "#fff";
    rr(k, X - 30, h - 22, 60, 30, 10);
    k.fill();
  };
