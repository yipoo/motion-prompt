Motion Prompt #080 · Highlighter animation
As the copy scrolls into view, a yellow highlighter stroke is drawn left to right across the lower half of the key words.

- Scroll: 0→90 px at 0.35–1.0 s, 90→180 px at 1.6–2.2 s, 180→0 at 3.3–4.4 s (cubic ease-in-out).
- Two lines of copy (20 px, bold) at 190 px and 290 px down the page, hidden below the view at first.
- When a line's top passes the trigger line (80% of the viewport height, 134 px), the stroke under its key words starts.
- Stroke: yellow #FFE36B, 11 tall, in the lower part of the text (8 px above to 3 px below the baseline). Width 0 → word width from the left over 0.8 s (cubic ease-out), drawn behind the text.
- Lines that leave the bottom of the view are reset out of sight.

Canvas: 539×208; cycle: 4.8s.

  // @element 08
    CELLS[7] = (k, w, h, u) => {
    frame(k);
    const SY = t => steps([ [ .35, 0 ], [ 1, 90 ], [ 1.6, 90 ], [ 2.2, 180 ], [ 3.3, 180 ], [ 4.4, 0 ] ], t), y = SY(u);
    const LINE = VH * .8, DUR = .8, MARK = "#FFE36B", F = `800 20px ${FONT}`;
    const rows = [ [ 190, [ MotionLocale.text("copy_090"), MotionLocale.text("copy_156"), MotionLocale.text("copy_095") ] ], [ 290, [ MotionLocale.text("copy_094"), MotionLocale.text("copy_147"), MotionLocale.text("copy_093") ] ] ];
    view(k, () => {
      k.save();
      k.translate(0, -y);
      txt(k, MotionLocale.text("copy_089"), 70, 50, `700 14px ${FONT}`, C.sub);
      txt(k, MotionLocale.text("copy_148"), 70, 80, `800 24px ${FONT}`, C.ink);
      bars(k, 70, 96, 330, 4, C.faint, 6, 12, .8);
      for (const [base, parts] of rows) {
        k.font = F;
        const ws = parts.map(s => k.measureText(MotionLocale.text(s)).width);
        let x = (VW - ws.reduce((a, b) => a + b, 0)) / 2;
        const t0 = enteredAt(SY, base - 20, LINE, u), q = t0 === null ? 0 : eOut(P(u, t0, t0 + DUR));
        if (q > 0) {
          k.fillStyle = MARK;
          k.fillRect(x + ws[0], base - 8, ws[1] * q, 11);
        }
        parts.forEach((s, i) => {
          txt(k, s, x, base, F, i === 1 ? C.ink : "#4A5368");
          x += ws[i];
        });
        bars(k, 110, base + 22, 291, 2, C.faint, 6, 12, .7);
      }
      k.restore();
      sbar(k, y / 180, .5);
    });
  };
