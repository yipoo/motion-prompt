Motion Prompt #089 · Preloader
On a navy screen the number under the logo counts from 0% to 100%; then the screen lifts away and the page underneath fades in.

- A navy screen fills the viewport with a logo, a number (28 px) and a bar (180 × 4).
- 0→100% at 0.25–2.1 s (cubic ease-out, rounded to whole numbers); the bar fills in orange at the same rate.
- At 2.3–2.85 s the whole screen moves up 168 px and out (cubic ease-in-out).
- The page heading and photo fade in 0→1 and rise from 14 px below at 2.35–2.95 s (cubic ease-out).
- At 4.25–4.75 s the screen comes back down from the top, at 0% (the next load).

Canvas: 539×208; cycle: 4.8s.

  // @element 17
    CELLS[16] = (k, w, h, u) => {
    frame(k);
    const n = Math.round(100 * eOut(P(u, .25, 2.1)));
    const cy = u < 4.25 ? -VH * eIO(P(u, 2.3, 2.85)) : -VH * (1 - eIO(P(u, 4.25, 4.75)));
    const show = eOut(P(u, 2.35, 2.95));
    view(k, () => {
      header(k, 0);
      k.save();
      k.globalAlpha = show;
      k.translate(0, 14 * (1 - show));
      txt(k, MotionLocale.text("copy_106"), 34, 82, `800 26px ${FONT}`, C.ink);
      bars(k, 34, 98, 200, 3);
      btn(k, 34, 136, 96, 24, MotionLocale.text("copy_100"));
      pic(k, 270, 46, 214, 112, 0, 8);
      k.restore();
      if (cy > -VH) {
        k.save();
        k.translate(0, cy);
        k.fillStyle = C.ink;
        k.fillRect(0, 0, VW, VH);
        logo(k, VW / 2 - 40, 56, "#fff");
        txt(k, (u >= 4.25 ? 0 : n) + "%", VW / 2, 98, `800 28px ${EN}`, "#fff", "center", "middle");
        k.fillStyle = "rgba(255,255,255,.15)";
        rr(k, VW / 2 - 90, 122, 180, 4, 2);
        k.fill();
        k.fillStyle = C.acc;
        rr(k, VW / 2 - 90, 122, 180 * (u >= 4.25 ? 0 : n / 100), 4, 2);
        k.fill();
        k.restore();
      }
    });
  };
