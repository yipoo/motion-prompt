Motion Prompt #095 · 动态渐变
标题背后，柔和的橙色、浅橙色、蓝色和浅蓝色色块绕圈漂移，不断以不同方式交融。

- 底色为 #FFF1EB，上方叠加四个径向渐变，均从中心颜色过渡到边缘透明，各自不透明度 0.85。
- 橙色半径 230，浅橙色 #FFB08F 半径 210，蓝色半径 240，浅蓝色 #9DB6FB 半径 200。
- 各中心每 4.8 s 沿椭圆运动一周；x 半径分别为 70、60、80、50，y 半径为 30、40、30、35，起始角为 0、2.1、4.2、1.0 rad。
- 白色标题（34 px）和按钮置于上方；只有渐变移动。

Canvas: 539×208; cycle: 4.8s.

  // @element 23
    CELLS[22] = (k, w, h, u) => {
    frame(k);
    const a = 2 * Math.PI * u / CYC;
    const BLOBS = [ [ C.acc, 150, 60, 230, 70, 30, 0 ], [ "#FFB08F", 380, 40, 210, 60, 40, 2.1 ], [ C.blue, 330, 150, 240, 80, 30, 4.2 ], [ "#9DB6FB", 90, 170, 200, 50, 35, 1 ] ];
    view(k, () => {
      k.fillStyle = "#FFF1EB";
      k.fillRect(0, 0, VW, VH);
      for (const [col, x0, y0, r, ax, ay, ph] of BLOBS) {
        const x = x0 + ax * Math.cos(a + ph), y = y0 + ay * Math.sin(a + ph);
        const g = k.createRadialGradient(x, y, 0, x, y, r);
        g.addColorStop(0, col);
        g.addColorStop(1, "rgba(255,255,255,0)");
        k.globalAlpha = .85;
        k.fillStyle = g;
        k.fillRect(0, 0, VW, VH);
        k.globalAlpha = 1;
      }
      txt(k, MotionLocale.text("Hello, Future."), 40, 82, `800 34px ${EN}`, "#fff", "left", "alphabetic", "1px");
      bars(k, 40, 100, 200, 2, "rgba(255,255,255,.8)", 6, 12, .6);
      btn(k, 40, 128, 104, 26, MotionLocale.text("copy_099"), "#fff", C.ink);
    });
  };
