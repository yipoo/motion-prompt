Motion Prompt #109 · 镜头眩光
太阳从建筑后露出时，周围出现光芒；光点、六边形和光环沿太阳穿过画面中心的直线延伸至另一侧。

- 场景 2：街角；y 132–162 的道路后方有建筑、树木和路灯（x 114、292、494）；人物站在左前方，双脚位于 x 176、y 200，身高 124 px。
- 太阳半径 13 px，从建筑后 (x 428, y 74) 移至开阔天空 (x 474, y 30)。
- 光芒 8 条，从距太阳中心 17 px 处向外延伸，终点交替为 26 px 或 44 px，即长度 9 px 和 27 px；琥珀色、粗 1.6 px、不透明度 75%。
- 眩光位于太阳穿过画面中心的直线上，t = 0.45、0.8、1.15、1.45、1.75、2.0，t = 1 为中心；依次是 7 px 圆、13 px 六边形、5 px 圆、22 px 六边形、9 px 圆和 30 px 光环（线宽 3 px）。
- 颜色为琥珀色 #FFC75A 和 #FFE3A3，不透明度为 40%–70% × 效果强度。
- 时序：效果在 0.4–1.4 s 渐入，保持到 4.0 s，并在 4.8 s 前消失；每轮 5 s，使用 cubic ease-in-out，约为 cubic-bezier(0.65,0,0.35,1)。

  // @element 13
    CELLS[12] = cell((k, u) => {
    const a = env(u), sx = lerp(428, 474, a), sy = lerp(74, 30, a);
    stageStreet(k, BASE, {
      sun: [ sx, sy, 13, 1 ]
    });
    const cx = CW / 2, cy = CH / 2, dx = cx - sx, dy = cy - sy;
    k.save();
    k.globalAlpha = .75 * a;
    k.strokeStyle = BASE.lit;
    k.lineWidth = 1.6;
    k.lineCap = "round";
    for (let i = 0; i < 8; i++) {
      const t = i / 8 * TAU + .2, L = i % 2 ? 26 : 44;
      k.beginPath();
      k.moveTo(sx + Math.cos(t) * 17, sy + Math.sin(t) * 17);
      k.lineTo(sx + Math.cos(t) * L, sy + Math.sin(t) * L);
      k.stroke();
    }
    const G = [ [ .45, 7, "c", .6 ], [ .8, 13, "h", .55 ], [ 1.15, 5, "c", .7 ], [ 1.45, 22, "h", .4 ], [ 1.75, 9, "c", .55 ], [ 2, 30, "r", .55 ] ];
    for (const [t, r, kind, al] of G) {
      const x = sx + dx * t, y = sy + dy * t;
      k.globalAlpha = al * a;
      if (kind === "r") {
        k.strokeStyle = BASE.lit;
        k.lineWidth = 3;
        k.beginPath();
        k.arc(x, y, r, 0, TAU);
        k.stroke();
        continue;
      }
      k.fillStyle = kind === "h" ? BASE.glow : BASE.lit;
      k.beginPath();
      if (kind === "h") for (let i = 0; i < 6; i++) {
        const t2 = i / 6 * TAU;
        k.lineTo(x + Math.cos(t2) * r, y + Math.sin(t2) * r);
      } else k.arc(x, y, r, 0, TAU);
      k.closePath();
      k.fill();
    }
    k.restore();
  });
