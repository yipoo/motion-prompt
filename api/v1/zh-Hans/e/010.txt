Motion Prompt #010 · 过冲回弹
仪表指针越过目标位置，回摆后逐渐稳定。

- 每 4.7 秒演示一次。半圆仪表（半径 128，描边 26）。目标为总弧度的 0.7（从左端起 126°）。
- 从 0.25 秒开始：数值 = 0.7 × (1 − e^(−4.6t)·cos(10.5t))。约 0.3 秒后超过目标约 25%，随后振荡并稳定在目标位置。
- 超出目标的部分以珊瑚色绘制。黄色刻度标记目标。
- 填充弧线为蓝色到薄荷色渐变；指针为白色，宽 9。
- 返回：从 3.7 秒起，数值 = 0.7 × e^(−7.5t)·cos(8.5t)，回到 0（停止前略低于 0）。

Canvas: 539×208; cycle: 4.7s.

  // @element 10
    CELLS[9] = (k, w, h, u) => {
    stage(k);
    const X = w / 2, Y = h - 26, R = 128, TG = .7;
    let v;
    if (u < .25) v = 0; else if (u < 3.7) {
      const t = u - .25;
      v = TG * (1 - Math.exp(-4.6 * t) * Math.cos(10.5 * t));
    } else {
      const t = u - 3.7;
      v = TG * Math.exp(-7.5 * t) * Math.cos(8.5 * t);
    }
    const ang = x => Math.PI + x * Math.PI;
    k.lineCap = "round";
    k.strokeStyle = C.navy2;
    k.lineWidth = 26;
    k.beginPath();
    k.arc(X, Y, R, ang(0), ang(1));
    k.stroke();
    if (v > .005) {
      const g = k.createLinearGradient(X - R, 0, X + R, 0);
      g.addColorStop(0, C.blue);
      g.addColorStop(1, C.mint);
      k.strokeStyle = g;
      k.beginPath();
      k.arc(X, Y, R, ang(0), ang(v));
      k.stroke();
    }
    if (v > TG + .004) {
      k.strokeStyle = C.coral;
      k.lineCap = "butt";
      k.beginPath();
      k.arc(X, Y, R, ang(TG), ang(v));
      k.stroke();
      k.lineCap = "round";
    }
    const ta = ang(TG);
    k.strokeStyle = C.yel;
    k.lineWidth = 7;
    k.beginPath();
    k.moveTo(X + Math.cos(ta) * (R - 22), Y + Math.sin(ta) * (R - 22));
    k.lineTo(X + Math.cos(ta) * (R + 30), Y + Math.sin(ta) * (R + 30));
    k.stroke();
    k.fillStyle = C.yel;
    k.beginPath();
    k.arc(X + Math.cos(ta) * (R + 38), Y + Math.sin(ta) * (R + 38), 8, 0, 7);
    k.fill();
    const a = ang(v);
    k.strokeStyle = "#fff";
    k.lineWidth = 9;
    k.beginPath();
    k.moveTo(X, Y);
    k.lineTo(X + Math.cos(a) * (R - 26), Y + Math.sin(a) * (R - 26));
    k.stroke();
    k.fillStyle = "#fff";
    k.beginPath();
    k.arc(X, Y, 17, 0, 7);
    k.fill();
    k.fillStyle = C.coral;
    k.beginPath();
    k.arc(X, Y, 8, 0, 7);
    k.fill();
  };
