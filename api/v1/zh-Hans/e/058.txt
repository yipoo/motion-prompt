Motion Prompt #058 · 轮播图
每次点击右箭头，三张不同图片组成的横排便向左移动一格，下方指示点随之变化。

- 542×191 卡片上的 4.5 秒周期。幻灯片尺寸 330×138，圆角半径 8，居中排列，间隔 18 px；相邻图片在边缘露出一部分。
- 左右各有带折角箭头的白色圆按钮（半径 18，阴影模糊 10）；下方三个指示点（当前项为蓝色 16×8，其余为 8×8）。
- 光标在 1.0、2.1 和 3.2 秒点击右箭头 (514, 88)。
- 每次点击后，整排在 0.4 秒内以 easeInOutCubic 向左移动 348 px（一张图片加间隔）。
- 三幅图片（风景、三个圆、柱状图）循环排列，因此移动三次后回到第一幅。

  // @element 10
    CELLS[9] = ui((w, h, t) => {
    const p = eio(seg(t, 1, 1.4)) + eio(seg(t, 2.1, 2.5)) + eio(seg(t, 3.2, 3.6));
    const sw = 330, gap = 18, cx = w / 2;
    X.save();
    X.beginPath();
    X.rect(0, 0, w, 160);
    X.clip();
    for (let j = Math.floor(p) - 2; j <= Math.floor(p) + 2; j++) {
      const x = cx - sw / 2 + (j - p) * (sw + gap), idx = (j % 3 + 3) % 3;
      if (idx === 0) mountains(x, 14, sw, 138);
      if (idx === 1) {
        fillRR(x, 14, sw, 138, 8, "#e3ebfd");
        circle(x + 90, 83, 34, C.mb);
        circle(x + 165, 83, 34, C.sb);
        circle(x + 240, 83, 34, C.mb);
      }
      if (idx === 2) {
        fillRR(x, 14, sw, 138, 8, C.lb);
        for (let k = 0; k < 5; k++) {
          const bh = [ 40, 70, 55, 95, 80 ][k];
          fillRR(x + 60 + k * 46, 132 - bh, 30, bh, 4, k === 3 ? C.blue : C.mb);
        }
      }
    }
    X.restore();
    [ [ 30, 83, Math.PI / 2 ], [ w - 30, 83, -Math.PI / 2 ] ].forEach(([x, y, r]) => {
      shadowed(() => circle(x, y, 18, "#fff"), 10, 3, .2);
      chevron(x, y, 5, r, "#555");
    });
    const act = Math.round(p) % 3;
    for (let k = 0; k < 3; k++) {
      const on = k === act;
      fillRR(w / 2 - 30 + k * 22 - (on ? 4 : 0), 172, on ? 16 : 8, 8, 4, on ? C.blue : C.line2);
    }
    pointer([ [ .3, w + 30, h + 30 ], [ .95, w - 28, 88 ], [ 3.4, w - 28, 88 ], [ 4.2, w + 30, h + 30 ] ], [ 1, 2.1, 3.2 ], t);
  });
