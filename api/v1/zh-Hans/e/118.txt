Motion Prompt #118 · 胶片颗粒
同一风景上覆盖细微颗粒，色彩略微偏向温暖的奶油色。

- 场景 3：左侧为半身像（头部中心 x 140、y 82，半径 28 px），右侧有山丘、房屋、树木和太阳（x 432、y 54，半径 15 px）。
- 颗粒：2,600 个方块，随机种子固定，尺寸 0.8–1.7 px；55% 为深色 #3A3530，45% 为白色。
- 颗粒不透明度为 16%–38% × 效果强度；颗粒保持不动，不闪烁。
- 颜色：每种颜色向 #F2DDBE 偏移 16%。
- 时序：效果在 0.4–1.4 s 渐入，保持到 4.0 s，并在 4.8 s 前消失；每轮 5 s，使用 cubic ease-in-out，约为 cubic-bezier(0.65,0,0.35,1)。

Canvas: 539×208; cycle: 5s.

  // @element 22
    CELLS[21] = cell((k, u) => {
    const a = env(u);
    const FILM = mapPal(BASE, c => mixc(c, "#F2DDBE", .16));
    const p = pal(BASE, FILM, a);
    const G = once("grain", () => {
      const r = rng(22), o = [];
      for (let i = 0; i < 2600; i++) o.push([ r() * CW, r() * CH, .8 + r() * .9, r() < .55 ? 0 : 1, .16 + r() * .22 ]);
      return o;
    });
    const grain = kk => {
      kk.save();
      for (const [x, y, s, d, al] of G) {
        kk.globalAlpha = al * a;
        kk.fillStyle = d ? "#FFFFFF" : "#3A3530";
        kk.fillRect(x, y, s, s);
      }
      kk.restore();
    };
    stageLand(k, p);
    grain(k);
  });
