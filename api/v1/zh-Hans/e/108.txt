Motion Prompt #108 · 追随拍摄
镜头跟随行驶的汽车；汽车在画面中央保持清晰，只有后方建筑、树木和路灯横向拖成条纹。

- 街角：y 132–162 的道路后方有建筑、树木和路灯，前方没有大人物。
- 汽车：无效果时在 5 s 内从 x −80 行驶至 620；有效果时保持在 x 270 ±10 px。
- 拖影：将背景横向分布绘制 12 次并取平均，分布范围从 0→46 px。
- 车轮：辐条线每 5 s 转动 12 圈。
- 右上角快门速度 1/1000→1/15。
- 时序：效果在 0.4–1.4 s 渐入，保持到 4.0 s，并在 4.8 s 前消失；每轮 5 s，使用 cubic ease-in-out，约为 cubic-bezier(0.65,0,0.35,1)。

  // @element 12
    CELLS[11] = cell((k, u) => {
    const a = env(u);
    const drive = lerp(-80, 620, u / CYC), x = lerp(drive, 270 + 10 * Math.sin(u / CYC * TAU), a);
    const S = 46 * a, n = S < .5 ? 1 : 12;
    for (let i = 0; i < n; i++) {
      k.save();
      k.globalAlpha = 1 / (i + 1);
      k.translate(n > 1 ? lerp(-S / 2, S / 2, i / (n - 1)) : 0, 0);
      stageStreet(k, BASE, {
        car: false,
        person: false
      });
      k.restore();
    }
    const D = pen(k);
    car(D, x, BASE, {
      head: BASE.lit
    });
    k.save();
    k.strokeStyle = BASE.tire;
    k.lineWidth = 1.6;
    for (const wx of [ x - 21, x + 21 ]) {
      const t = u / CYC * TAU * 12;
      k.beginPath();
      k.moveTo(wx - Math.cos(t) * 5, 156 - Math.sin(t) * 5);
      k.lineTo(wx + Math.cos(t) * 5, 156 + Math.sin(t) * 5);
      k.stroke();
    }
    k.restore();
    hud(k, a > .5 ? "1/15" : "1/1000");
  });
