Motion Prompt #094 · 跑马灯
藏青色横带内，大号服务名称以橙色圆点分隔，持续无缝向左移动。

- 横带高 58、藏青色、宽度铺满视口。文字为白色，30 px、特粗、字距 1 px。
- 一组包含四个词：WEB DESIGN / BRANDING / MOTION / UI/UX，间隔 64 px，每个空隙有一个橙点（半径 6）。
- 每个循环（4.8 s）以恒定速度向左移动一组内容的宽度，此处约 881 px。先测量该组宽度，再按此宽度循环偏移，因此看不见接缝。
- 重复该组内容，直到填满视口宽度。

Canvas: 539×208; cycle: 4.8s.

  // @element 22
    CELLS[21] = (k, w, h, u) => {
    frame(k);
    const WORDS = [ MotionLocale.text("WEB DESIGN"), MotionLocale.text("BRANDING"), MotionLocale.text("MOTION"), MotionLocale.text("UI/UX") ], F = `800 30px ${EN}`, GAPW = 64;
    view(k, () => {
      bars(k, 34, 20, 180, 1, C.ink, 10);
      bars(k, 34, 38, 300, 1);
      k.fillStyle = C.ink;
      k.fillRect(0, 58, VW, 58);
      k.font = F;
      k.letterSpacing = "1px";
      const ws = WORDS.map(s => k.measureText(MotionLocale.text(s)).width);
      k.letterSpacing = "0px";
      const unit = ws.reduce((a, b) => a + b + GAPW, 0);
      let x = -mod(u / CYC, 1) * unit;
      for (let rep = 0; x < VW; rep++) WORDS.forEach((s, i) => {
        txt(k, s, x, 88, F, "#fff", "left", "middle", "1px");
        k.fillStyle = C.acc;
        k.beginPath();
        k.arc(x + ws[i] + GAPW / 2, 88, 6, 0, 7);
        k.fill();
        x += ws[i] + GAPW;
      });
      for (let i = 0; i < 3; i++) {
        k.fillStyle = C.soft;
        rr(k, 34 + i * 152, 130, 136, 30, 6);
        k.fill();
        bars(k, 46 + i * 152, 142, 90, 1, C.faint);
      }
    });
  };
