Motion Prompt #189 · 切出镜头
从讲话者切到聆听的观众，再切回来；插入的是不同主体。

- 衔接：两次直接切换，过渡为 0 帧；1.7 s 切至另一主体，3.0 s 返回原场景，切出持续 1.3 s。
- 主镜头 A：相机固定，高度 1.4 m；前方 5.5 m 的人物讲话，四肢按 0.45·sin(u×6) rad 摆动，右侧 0.9 m 有灰色讲台。
- 另一主体 CUTAWAY：聆听的观众，五人位于 3.1–4.6 m 远处，呈蓝色和浅蓝色，举起双臂，按 |sin(t×9+i)|×5 cm 上下弹动；没有背景，代表不同地点。
- 底部条带：A 为灰色、另一主体为蓝色、再回到灰色 A，并带播放头（17 Insert shot 展示同一场景的细节）。
- 右上角显示 SHOT A→CUTAWAY→SHOT A；4.7 s 时直接切回起点。

  // @element 21
    const CUT = [ 1.7, 3 ];
  CELLS[20] = shot((k, w, h, u) => {
    const away = u >= CUT[0] && u < CUT[1];
    if (!away) world(k, w, h, cam({
      y: 1.4
    }), {
      person: {
        x: 0,
        z: 5.5,
        walk: .45 * Math.sin(u * 6)
      },
      extra: [ [ .9, 0, 5.6, kk => {
        kk.fillStyle = C.g3;
        kk.fillRect(-.3, 0, .6, 1.1);
        kk.fillRect(-.4, 1.05, .8, .1);
      } ] ]
    }); else {
      const t = u - CUT[0], ex = [];
      [ [ -1.6, 3.2 ], [ -.2, 3.6 ], [ 1.3, 3.1 ], [ .6, 4.4 ], [ -1.1, 4.6 ] ].forEach(([x, z], i) => ex.push([ x, 0, z, kk => {
        kk.save();
        kk.translate(0, Math.abs(Math.sin(t * 9 + i)) * .05);
        person(kk, 1, {
          armUp: true,
          col: i % 2 ? C.blue : C.blueM
        });
        kk.restore();
      } ]));
      world(k, w, h, cam({
        y: 1.6,
        pitch: -.08
      }), {
        buildings: false,
        trees: false,
        mountains: false,
        sun: null,
        extra: ex
      });
    }
    strip(k, w, h, [ [ 0, CUT[0], "A", C.g2 ], [ CUT[0], CUT[1], MotionLocale.text("copy_056"), C.blue ], [ CUT[1], CYC, "A", C.g2 ] ], u);
    return away ? MotionLocale.text("CUTAWAY") : MotionLocale.text("SHOT A");
  });
