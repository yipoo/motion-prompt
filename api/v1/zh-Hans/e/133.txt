Motion Prompt #133 · 文字遮罩
粗体文字形成镂空，流动的海景只在字内显现；周期接近结束时显示完整画面。

- 文字 OCEAN，字重 900，system-ui，字距 2 px。字号从 172 px 起，每次减小 2 px，直至单词宽度不超过 500 px。
- 内部景色持续运动：珊瑚色天空、上下浮动 ±6 px 的黄色太阳、y = 116 以下的蓝色海面；四条白色波浪线每 4.7 s 交替向相反方向漂移 240 px；帆船每 4.7 s 横穿一次，云漂移 60 px。
- 实现方式：在离屏画布绘制文字，再用 source-in 合成景色，使景色只保留在字形内部。文字外沿有 4 px 白色描边。
- 文字外部景色的不透明度为 12%；3.45–3.85 s 升至 100%，4.25–4.65 s 恢复到 12%（smoothstep）。

  function seaPattern(g, w, h, u) {
    g.fillStyle = C.coral;
    g.fillRect(0, 0, w, h);
    g.fillStyle = C.yel;
    g.beginPath();
    g.arc(w * .6, 116 + 6 * Math.sin(2 * Math.PI * u / CYC), 60, 0, 7);
    g.fill();
    g.fillStyle = C.blue;
    g.fillRect(0, 116, w, h - 116);
    const off = u / CYC * 240;
    g.strokeStyle = "#fff";
    g.lineWidth = 7;
    g.lineCap = "round";
    for (let j = 0; j < 4; j++) {
      const y = 128 + j * 22, d = j % 2 ? -1 : 1;
      g.beginPath();
      for (let x = -20; x <= w + 20; x += 6) {
        const yy = y + 5 * Math.sin((x + off * d + j * 40) * 2 * Math.PI / 120);
        x === -20 ? g.moveTo(x, yy) : g.lineTo(x, yy);
      }
      g.stroke();
    }
    const bx = mod(u / CYC * (w + 160), w + 160) - 80;
    g.fillStyle = "#fff";
    g.beginPath();
    g.moveTo(bx, 46);
    g.lineTo(bx, 100);
    g.lineTo(bx + 40, 100);
    g.closePath();
    g.fill();
    g.fillStyle = C.navy;
    g.beginPath();
    g.moveTo(bx - 42, 104);
    g.lineTo(bx + 52, 104);
    g.lineTo(bx + 38, 122);
    g.lineTo(bx - 30, 122);
    g.closePath();
    g.fill();
    g.fillStyle = "#fff";
    [ [ 60, 40 ], [ 320, 30 ], [ 470, 56 ] ].forEach(([x, y]) => {
      const xx = mod(x + u / CYC * 60, w + 80) - 40;
      g.beginPath();
      g.ellipse(xx, y, 30, 11, 0, 0, 7);
      g.fill();
    });
  }
