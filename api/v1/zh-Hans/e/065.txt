Motion Prompt #065 · 汉堡菜单
点击页头右侧的三横线图标后，图标变为 ×，菜单从页头下方下拉；再次点击则关闭。

- 542×191 卡片，循环 4.5 s。白色页头高 44；右侧 (500, 22) 处放置三条线，长 24、粗 3、间距 8 px。
- 光标在 1.0 s 和 3.3 s 点击。展开程度 a：1.0–1.3 s 从 0→1，3.3–3.6 s 从 1→0，使用 easeInOutCubic。
- 三横线变为 ×：上下两条线向中央移动 8 px，同时分别旋转 ±45°；中间线以不透明度 1 − a 淡出。展开期间显示蓝色。
- 菜单为白色，高 140，顶部 y = 44 − 140 + 140·a，因此从页头下方下拉（只显示页头以下的部分）；阴影模糊 16，内含四项，第一项为蓝色。

  // @element 17
    CELLS[16] = ui((w, h, t) => {
    const a = eio(seg(t, 1, 1.3)) * (1 - eio(seg(t, 3.3, 3.6)));
    pageLines(24, 66, 494);
    fillRR(24, 134, 494, 46, 8, C.block);
    if (a > 0) {
      X.save();
      X.beginPath();
      X.rect(0, 44, w, h);
      X.clip();
      const mh = 140, my = 44 - mh + mh * a;
      shadowed(() => {
        X.fillStyle = "#fff";
        X.fillRect(0, my, w, mh);
      }, 16, 6, .16);
      for (let k = 0; k < 4; k++) {
        const y = my + 14 + k * 32;
        bar(24, y + 6, [ 110, 150, 90, 130 ][k], k === 0 ? C.blue : C.line2, 11);
        if (k < 3) {
          X.fillStyle = C.line;
          X.fillRect(24, y + 26, w - 48, 1);
        }
      }
      X.restore();
    }
    X.fillStyle = "#fff";
    X.fillRect(0, 0, w, 44);
    X.fillStyle = C.line;
    X.fillRect(0, 44, w, 1.5);
    fillRR(24, 15, 76, 16, 4, C.mb);
    const hx = w - 42, hy = 22;
    X.save();
    X.translate(hx, hy);
    X.strokeStyle = a > .02 ? C.blue : "#333";
    X.lineWidth = 3;
    X.lineCap = "round";
    X.save();
    X.translate(0, -8 * (1 - a));
    X.rotate(Math.PI / 4 * a);
    X.beginPath();
    X.moveTo(-12, 0);
    X.lineTo(12, 0);
    X.stroke();
    X.restore();
    X.globalAlpha = 1 - a;
    X.beginPath();
    X.moveTo(-12, 0);
    X.lineTo(12, 0);
    X.stroke();
    X.globalAlpha = 1;
    X.save();
    X.translate(0, 8 * (1 - a));
    X.rotate(-Math.PI / 4 * a);
    X.beginPath();
    X.moveTo(-12, 0);
    X.lineTo(12, 0);
    X.stroke();
    X.restore();
    X.restore();
    pointer([ [ .3, w + 30, h + 30 ], [ .95, w - 38, 26 ], [ 3.5, w - 38, 26 ], [ 4.2, w + 30, h + 30 ] ], [ 1, 3.3 ], t);
  });
