Motion Prompt #021 · 翻转
卡片绕垂直轴翻转，露出另一面。

- 卡片（300 × 156，圆角半径 22）。正面（蓝色，Q）⇄ 背面（黄色，A），每面 2.35 秒。
- 0.45–1.5 秒（1.05 秒）以 backOut（s = 1.5）旋转 180°，略微超过 180° 后回正。
- 卡片宽度 = 原始宽度 × |cos θ|。经过 90° 时替换卡面图案。
- 透视：将卡片绘制成细竖条，从一侧到另一侧，各条高度最多变化 ±20% × sin θ（远侧更短，近侧更高）。
- 翻转时，按正弦曲线放大 8% 并上浮 6 px；卡面覆盖不透明度为 0.34 × |sin θ| 的暗色遮层。地面上有椭圆阴影。

Canvas: 539×208; cycle: 4.7s.

  // @element 21
    const FW = 300, FH = 156, FC = mkCanvas(FW, FH), FK = FC.getContext("2d");
  function flipFace(i) {
    FK.setTransform(1, 0, 0, 1, 0, 0);
    FK.clearRect(0, 0, FW, FH);
    FK.fillStyle = i ? C.yel : C.blue;
    rr(FK, 0, 0, FW, FH, 22);
    FK.fill();
    FK.strokeStyle = i ? "rgba(15,26,61,.18)" : "rgba(255,255,255,.3)";
    FK.lineWidth = 4;
    rr(FK, 12, 12, FW - 24, FH - 24, 13);
    FK.stroke();
    txt(FK, i ? "A" : "Q", 92, FH / 2 + 4, `800 112px ${DISP}`, i ? C.navy : "#fff", "center", "middle");
    FK.fillStyle = i ? C.navy : "#fff";
    FK.globalAlpha = i ? .85 : .9;
    rr(FK, 164, 50, 100, 14, 7);
    FK.fill();
    FK.globalAlpha = i ? .4 : .5;
    rr(FK, 164, 76, 76, 12, 6);
    FK.fill();
    rr(FK, 164, 98, 90, 12, 6);
    FK.fill();
    FK.globalAlpha = 1;
  }
  CELLS[20] = abCell((k, w, h, u, a, b) => {
    stage(k);
    const pr = P(u, TA - .05, TB), th = Math.PI * backOut(pr, 1.5), c = Math.cos(th), s = Math.sin(th), front = c >= 0;
    flipFace(front ? a : b);
    const X = w / 2, Y = h / 2 - 2, lift = Math.sin(Math.PI * pr), sc = 1 + .08 * lift, ww = FW * Math.abs(c) * sc, per = .2 * s * (front ? 1 : -1);
    k.fillStyle = "rgba(0,0,0,.3)";
    k.beginPath();
    k.ellipse(X, h - 13, Math.max(8, ww / 2 * .9), 6, 0, 0, 7);
    k.fill();
    const STEP = 2;
    for (let sx = 0; sx < FW; sx += STEP) {
      const f = sx / FW, dx = X - ww / 2 + f * ww, hh = FH * sc * (1 + per * (f - .5) * 2);
      k.drawImage(FC, sx, 0, STEP, FH, dx, Y - hh / 2 - 6 * lift, ww / FW * STEP + .6, hh);
    }
    if (Math.abs(s) > .02) {
      k.fillStyle = `rgba(8,14,40,${.34 * Math.abs(s)})`;
      k.beginPath();
      const hl = FH * sc * (1 - per), hr = FH * sc * (1 + per), yy = Y - 6 * lift;
      k.moveTo(X - ww / 2, yy - hl / 2);
      k.lineTo(X + ww / 2, yy - hr / 2);
      k.lineTo(X + ww / 2, yy + hr / 2);
      k.lineTo(X - ww / 2, yy + hl / 2);
      k.closePath();
      k.fill();
    }
  });
