Motion Prompt #127 · 往返循环
方块移动到一端后，像倒放一样沿原路返回，如此反复。

- 每 4.7 s 往返一次：正向 2.35 s，再将相同动作反向播放 2.35 s；进度 g 线性变化 0→1→0。
- 薄荷绿圆角方块，边长 48 px、圆角 13，从 x = 92 跳到 447，沿三个高度为 62·|sin(3πg)| 的拱形轨迹移动，并随 g 旋转一整圈。
- 四个残影副本，相隔 0.045 s，不透明度 0.1–0.4。
- 下方有播放条：正向时为蓝色并标注播放，反向时为珊瑚色并标注倒放。进度条与旋钮跟随 g。
- 竖柱标记两个折返点。

  // @element 07
    CELLS[6] = (k, w, h, u) => {
    u = snapU(u);
    stage(k);
    const q = u / CYC, fwd = q < .5, g = fwd ? 2 * q : 2 - 2 * q, X0 = 92, X1 = w - 92, base = 136, S = 48;
    const pos = gg => [ lerp(X0, X1, gg), base - 62 * Math.abs(Math.sin(3 * Math.PI * gg)) ];
    k.save();
    k.setLineDash([ 2, 12 ]);
    k.lineCap = "round";
    k.strokeStyle = C.navy3;
    k.lineWidth = 5;
    k.beginPath();
    for (let i = 0; i <= 150; i++) {
      const [px, py] = pos(i / 150);
      i ? k.lineTo(px, py - S / 2) : k.moveTo(px, py - S / 2);
    }
    k.stroke();
    k.restore();
    k.fillStyle = C.navy3;
    rr(k, X0 - 40, base + 2, X1 - X0 + 80, 5, 2.5);
    k.fill();
    [ X0, X1 ].forEach(xx => {
      k.fillStyle = C.navy3;
      rr(k, xx - 4, base - 70, 8, 72, 4);
      k.fill();
    });
    for (let j = 4; j >= 1; j--) {
      const qq = mod(u - j * .045, CYC) / CYC, gg = qq < .5 ? 2 * qq : 2 - 2 * qq, [gx, gy] = pos(gg);
      k.globalAlpha = .1 * (5 - j);
      k.fillStyle = C.mint;
      k.save();
      k.translate(gx, gy - S / 2);
      k.rotate(gg * Math.PI * 2);
      rr(k, -S / 2, -S / 2, S, S, 13);
      k.fill();
      k.restore();
    }
    k.globalAlpha = 1;
    const [x, y] = pos(g);
    k.save();
    k.translate(x, y - S / 2);
    k.rotate(g * Math.PI * 2);
    k.fillStyle = C.mint;
    rr(k, -S / 2, -S / 2, S, S, 13);
    k.fill();
    k.fillStyle = "#fff";
    k.beginPath();
    k.arc(S / 2 - 13, -S / 2 + 13, 6.5, 0, 7);
    k.fill();
    k.restore();
    const ty = 184, tx0 = 196, tx1 = w - 34;
    k.fillStyle = fwd ? C.blue : C.coral;
    rr(k, 22, ty - 20, 158, 40, 20);
    k.fill();
    txt(k, fwd ? MotionLocale.text("copy_012") : MotionLocale.text("copy_013"), 101, ty + 1, `800 25px ${FONT}`, "#fff", "center", "middle");
    k.fillStyle = C.navy3;
    rr(k, tx0, ty - 5, tx1 - tx0, 10, 5);
    k.fill();
    k.fillStyle = fwd ? C.blue : C.coral;
    rr(k, tx0, ty - 5, Math.max(10, (tx1 - tx0) * g), 10, 5);
    k.fill();
    k.fillStyle = C.yel;
    k.beginPath();
    k.arc(tx0 + (tx1 - tx0) * g, ty, 12, 0, 7);
    k.fill();
  };
