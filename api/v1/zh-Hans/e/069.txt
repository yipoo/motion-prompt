  // @element 21
    CELLS[20] = ui((w, h, t) => {
    const sk = seg(t, 1.05, 1.2) * (1 - seg(t, 3, 3.3));
    ring(w - 40, 26, 14, C.line2, 1.8);
    X.beginPath();
    X.arc(w - 40, 26, 7, -Math.PI * .1, Math.PI * 1.4);
    X.strokeStyle = "#666";
    X.lineWidth = 2.2;
    X.stroke();
    bar(24, 20, 160, C.line2, 12);
    for (let k = 0; k < 2; k++) {
      const y = 56 + k * 66;
      X.save();
      X.globalAlpha = 1 - sk;
      mountains(24, y, 96, 56, k);
      bar(136, y + 6, 200, C.ink, 11);
      bar(136, y + 26, 340);
      bar(136, y + 42, 260);
      X.restore();
      if (sk > 0) {
        X.save();
        X.globalAlpha = sk;
        const g = "#e9e7e5";
        fillRR(24, y, 96, 56, 8, g);
        fillRR(136, y + 4, 200, 14, 7, g);
        fillRR(136, y + 26, 340, 10, 5, g);
        fillRR(136, y + 42, 260, 10, 5, g);
        const sx = -200 + (t - 1.1) % .9 / .9 * (w + 400);
        X.globalCompositeOperation = "source-atop";
        const gr = X.createLinearGradient(sx - 80, 0, sx + 80, 0);
        gr.addColorStop(0, "rgba(255,255,255,0)");
        gr.addColorStop(.5, "rgba(255,255,255,.75)");
        gr.addColorStop(1, "rgba(255,255,255,0)");
        X.fillStyle = gr;
        X.fillRect(0, y - 2, w, 64);
        X.restore();
      }
    }
    pointer([ [ .3, w + 30, h + 30 ], [ .95, w - 38, 30 ], [ 1.4, w - 38, 30 ], [ 2.1, w + 30, h + 30 ] ], [ 1 ], t);
  });
