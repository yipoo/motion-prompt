Motion Prompt #004 · 线条描绘
图表曲线逐渐绘出，随后从起点一端逐渐擦除。

- 每 4.7 秒演示一次。采用路径修剪：为每条描边的可见范围（起始比例到结束比例）设置动画。
- 坐标轴（L 形，线宽 6）在 0.08–0.7 秒绘出；曲线（薄荷色，线宽 11，平滑穿过 7 个点，圆形端点）在 0.4–1.95 秒绘出。两者均使用三次方缓入缓出。
- 绘制时，一个黄色圆点（半径 12，白色内圆半径 5）随线条前端移动。
- 箭头：1.9–2.2 秒以 backOut（s = 2.5）展开；两翼长 26、宽 20。
- 珊瑚色圆圈（半径 26，线宽 7）在 2.2–2.8 秒绕某个峰值绘制，从顶部顺时针画出。
- 擦除：圆圈 3.5–4.0 秒、曲线 3.65–4.45 秒、坐标轴 3.85–4.55 秒，均从起点一端收缩。箭头在 4.3–4.45 秒淡出。

    function smoothPath(pts, n = 16) {
    const out = [];
    for (let i = 0; i < pts.length - 1; i++) {
      const p0 = pts[Math.max(0, i - 1)], p1 = pts[i], p2 = pts[i + 1], p3 = pts[Math.min(pts.length - 1, i + 2)];
      for (let j = 0; j < n; j++) {
        const t = j / n, t2 = t * t, t3 = t2 * t;
        out.push([ .5 * (2 * p1[0] + (-p0[0] + p2[0]) * t + (2 * p0[0] - 5 * p1[0] + 4 * p2[0] - p3[0]) * t2 + (-p0[0] + 3 * p1[0] - 3 * p2[0] + p3[0]) * t3), .5 * (2 * p1[1] + (-p0[1] + p2[1]) * t + (2 * p0[1] - 5 * p1[1] + 4 * p2[1] - p3[1]) * t2 + (-p0[1] + 3 * p1[1] - 3 * p2[1] + p3[1]) * t3) ]);
      }
    }
    out.push(pts[pts.length - 1]);
    return out;
  }
  function pathLen(pts) {
    const c = [ 0 ];
    for (let i = 1; i < pts.length; i++) c.push(c[i - 1] + Math.hypot(pts[i][0] - pts[i - 1][0], pts[i][1] - pts[i - 1][1]));
    return c;
  }
  function ptAt(pts, cum, f) {
    const L = cum[cum.length - 1] * clamp(f);
    let i = 1;
    while (i < cum.length - 1 && cum[i] < L) i++;
    const t = (L - cum[i - 1]) / Math.max(1e-6, cum[i] - cum[i - 1]);
    return [ lerp(pts[i - 1][0], pts[i][0], t), lerp(pts[i - 1][1], pts[i][1], t), Math.atan2(pts[i][1] - pts[i - 1][1], pts[i][0] - pts[i - 1][0]) ];
  }
  function trim(k, pts, cum, f0, f1) {
    if (f1 - f0 <= 5e-4) return;
    const L = cum[cum.length - 1], a = L * f0, b = L * f1;
    k.beginPath();
    const s = ptAt(pts, cum, f0);
    k.moveTo(s[0], s[1]);
    for (let i = 1; i < pts.length - 1; i++) if (cum[i] > a && cum[i] < b) k.lineTo(pts[i][0], pts[i][1]);
    const e = ptAt(pts, cum, f1);
    k.lineTo(e[0], e[1]);
    k.stroke();
  }
