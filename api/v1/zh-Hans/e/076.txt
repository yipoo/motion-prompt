Motion Prompt #076 · 滚动固定
藏青色区块占满视口后固定不动；继续滚动会让数字与照片依次切换 1→2→3，之后区块继续滚走。

- 滚动量：0.3–3.0 s 为 0→500 px，3.5–4.5 s 为 500→0（cubic ease-in-out）。
- 区块高 168、藏青色，在滚动 110 px 时到达顶部，并在接下来的 300 px 滚动中保持固定（GSAP ScrollTrigger pin）。
- 固定期间的进度 p = (滚动量 − 110) / 300。左侧长 124 的竖线按 p 填充橙色。
- 在 p = 1/3 和 2/3 时切换下一步（±0.07，cubic ease-in-out）：数字（46 px）向上滚动 52 px，照片（226×124）向上滑动 124 px。
- 经过 300 px 后解除固定，区块与其下的白色区块一起向上滚动。

Canvas: 539×208; cycle: 4.8s.

  // @element 04
    CELLS[3] = (k, w, h, u) => {
    frame(k);
    const y = steps([ [ .3, 0 ], [ 3, 500 ], [ 3.5, 500 ], [ 4.5, 0 ] ], u);
    const PIN0 = 110, PD = 300;
    const top = y < PIN0 ? PIN0 - y : y < PIN0 + PD ? 0 : PIN0 + PD - y, p = clamp((y - PIN0) / PD);
    const f = eIO(P(p, 1 / 3 - .07, 1 / 3 + .07)) + eIO(P(p, 2 / 3 - .07, 2 / 3 + .07));
    view(k, () => {
      txt(k, MotionLocale.text("copy_155"), 34, 56 - y, `800 22px ${FONT}`, C.ink);
      bars(k, 34, 72 - y, 240, 2);
      k.save();
      k.translate(0, top);
      k.fillStyle = C.ink;
      k.fillRect(0, 0, VW, VH);
      txt(k, MotionLocale.text("STEP"), 34, 40, `700 12px ${EN}`, "rgba(255,255,255,.6)", "left", "alphabetic", "2px");
      k.save();
      k.beginPath();
      k.rect(30, 46, 120, 52);
      k.clip();
      for (let i = 0; i < 3; i++) txt(k, "0" + (i + 1), 32, 90 + (i - f) * 52, `800 46px ${EN}`, "#fff");
      k.restore();
      bars(k, 34, 110, 150, 3, "rgba(255,255,255,.28)");
      k.fillStyle = "rgba(255,255,255,.16)";
      rr(k, 232, 22, 4, 124, 2);
      k.fill();
      k.fillStyle = C.acc;
      rr(k, 232, 22, 4, 124 * p, 2);
      k.fill();
      k.save();
      rr(k, 256, 22, 226, 124, 8);
      k.clip();
      for (let i = 0; i < 3; i++) pic(k, 256, 22 + (i - f) * 124, 226, 124, i, 0);
      k.restore();
      k.fillStyle = "#fff";
      k.fillRect(0, VH, VW, 200);
      bars(k, 34, VH + 30, 300, 3);
      k.restore();
      sbar(k, y / 500, .3);
    });
  };
