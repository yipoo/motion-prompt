Motion Prompt #035 · FPV 穿越机镜头
以第一人称视角在两排树之间低空高速飞行，机身左右倾斜。

- 相机：恒定以 14 m/s 向前飞行（显示为 50km/h），无缓动；俯仰角固定为 −0.05 rad。
- 穿行摆动（T = 2π×u/4.7，每 4.7 秒摆动两次）：横向位移 0.6·sin(2T-0.4) m，高度 1.4 + 0.5·sin(2T) m。
- 倾侧：横滚角 0.22·sin(2T+0.6) rad（约 ±12.6°），偏航角 0.05·sin(2T) rad（约 ±2.9°）。
- 无主体。两侧距离各 3.6 m 处有高 4.2 m 的树，每隔 8 m 一棵，两排前后错开 4 m，向相机快速冲来。
- 速度线：10 条长约 32–76 px 的半透明灰色放射线（靠近两侧更长），向外流动并循环。
- 无停留，也不以交叉淡化重置（每 4.7 秒重新开始时，树列相位发生偏移）。

    CELLS[10] = shot((k, w, h, u) => {
    const sp = 14, z = u * sp, T = u * 2 * Math.PI / CYC;
    const c = cam({
      z: z,
      y: 1.4 + .5 * Math.sin(T * 2),
      roll: .22 * Math.sin(T * 2 + .6),
      yaw: .05 * Math.sin(T * 2),
      x: .6 * Math.sin(T * 2 - .4),
      pitch: -.05
    });
    const ex = [];
    const L = 8, base = Math.floor(z / L) * L;
    for (let i = 0; i < 12; i++) {
      const zz = base + i * L;
      ex.push([ -3.6, 0, zz, (kk, s) => tree(kk, s, 4.2) ]);
      ex.push([ 3.6, 0, zz + L / 2, (kk, s) => tree(kk, s, 4.2, C.blueXL) ]);
    }
    world(k, w, h, c, {
      buildings: false,
      trees: false,
      extra: ex,
      gridX: 20,
      after: kk => {
        kk.strokeStyle = "rgba(150,150,155,.45)";
        kk.lineWidth = 2;
        for (let i = 0; i < 10; i++) {
          const a = i / 10 * Math.PI * 2 + .3, ph = mod(u * 2.2 + i * .37, 1);
          const r0 = 120 + ph * 160;
          kk.beginPath();
          kk.moveTo(Math.cos(a) * r0 * 1.9, Math.sin(a) * r0 * .8);
          kk.lineTo(Math.cos(a) * (r0 + 40) * 1.9, Math.sin(a) * (r0 + 40) * .8);
          kk.stroke();
        }
      }
    });
    return `${Math.round(sp * 3.6)}km/h`;
  });
