页面从角落卷起并翻折，显露下面的下一页。

- 第 1 页（白色）→第 2 页（奶油色）→第 1 页，每段 2.35 s。0.35–1.75 s（持续 1.4 s）从右下角卷起，二次缓入缓出。
- 折线从角落沿归一化方向 (−1, −0.42) 移动 0→600 px。
- 折线外侧部分沿折线镜像，形成翻折部分，即浅灰或米色的纸张背面，并添加投影（模糊 14，偏移 4、4）及 2 px 白色边线。
- 纸张抬起的区域显露下一页，沿折线带有 50 px 宽的渐变阴影。

  function clipHalf(poly, d, s0, keepLess) {
    const out = [], f = p => (p[0] - CW) * d[0] + (p[1] - CH) * d[1] - s0;
    for (let i = 0; i < poly.length; i++) {
      const A = poly[i], B = poly[(i + 1) % poly.length], fa = f(A), fb = f(B), ia = keepLess ? fa <= 0 : fa >= 0, ib = keepLess ? fb <= 0 : fb >= 0;
      if (ia) out.push(A);
      if (ia !== ib) {
        const t = fa / (fa - fb);
        out.push([ lerp(A[0], B[0], t), lerp(A[1], B[1], t) ]);
      }
    }
    return out;
  }
