Motion Prompt #111 · 色差
街景分解为红、绿、蓝三层，以略有不同的尺寸叠回，人物、建筑和路灯边缘出现红色与青色彩边，越靠两侧越明显。

- 场景 2：街角；y 132–162 的道路后方有建筑、树木和路灯（x 114、292、494）；人物站在左前方，双脚位于 x 176、y 200，身高 124 px。
- 分色：图像分别与红色 #FF0000、绿色 #00FF00、蓝色 #0000FF 相乘生成三个副本，再在黑色底上相加合成。
- 以中心缩放：红色 1 + 2s，绿色 1 + s，蓝色 1，其中 s = 0→0.02。
- 偏移：左右边缘处红蓝两层最终相距约 11 px，中央为 0。
- 时序：效果在 0.4–1.4 s 渐入，保持到 4.0 s，并在 4.8 s 前消失；每轮 5 s，使用 cubic ease-in-out，约为 cubic-bezier(0.65,0,0.35,1)。

Canvas: 539×208; cycle: 5s.

  // @element 15
    CELLS[14] = cell((k, u) => {
    const a = env(u), s = .02 * a, cx = CW / 2, cy = CH / 2;
    if (s < 5e-4) {
      stageStreet(k, BASE);
      return;
    }
    const [S, ks] = layer("ca");
    stageStreet(ks, BASE);
    k.fillStyle = "#000000";
    k.fillRect(0, 0, CW, CH);
    for (const [col, sc] of [ [ "#FF0000", 1 + 2 * s ], [ "#00FF00", 1 + s ], [ "#0000FF", 1 ] ]) {
      const [L, kl] = layer("ca-" + col);
      kl.drawImage(S, 0, 0);
      kl.globalCompositeOperation = "multiply";
      kl.fillStyle = col;
      kl.fillRect(0, 0, CW, CH);
      kl.globalCompositeOperation = "source-over";
      k.save();
      k.globalCompositeOperation = "lighter";
      k.translate(cx, cy);
      k.scale(sc, sc);
      k.translate(-cx, -cy);
      k.drawImage(L, 0, 0);
      k.restore();
    }
  });
