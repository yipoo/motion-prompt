Motion Prompt #131 · 路径文字
一行文字沿波浪线流动，每个字母随路径方向转动。

- 路径：y = 118 + 42·sin(2πx ÷ 330 + 0.6)，x 从 −70 到 609；在下方 6 px 处显示藏青色虚线（线段/间隔 14/10，线宽 5）。
- 重复排列 ALONG THE LINE • （字号 42 px，粗体，字距 2 px）。每个字符按弧长定位在路径上，并旋转至路径方向。
- 文字每 4.7 s 沿路径前进一个完整字符串的长度，使循环无缝衔接。
- 字母为白色，圆点为黄色。

Canvas: 539×208; cycle: 4.7s.

  // @element 11
    const TOP = (() => {
    const pts = [];
    for (let x = -70; x <= CW + 70; x += 4) pts.push([ x, 118 + 42 * Math.sin(x * 2 * Math.PI / 330 + .6) ]);
    return {
      pts: pts,
      cum: pathLen(pts)
    };
  })();
  CELLS[10] = (k, w, h, u) => {
    stage(k);
    k.save();
    k.setLineDash([ 14, 10 ]);
    k.strokeStyle = C.navy3;
    k.lineWidth = 5;
    k.lineCap = "round";
    k.beginPath();
    TOP.pts.forEach((p, i) => i ? k.lineTo(p[0], p[1] + 6) : k.moveTo(p[0], p[1] + 6));
    k.stroke();
    k.restore();
    const f = `800 42px ${DISP}`, S = MotionLocale.text("ALONG THE LINE • ");
    k.font = f;
    const ws = [ ...S ].map(ch => k.measureText(ch).width + 2), run = ws.reduce((a, b) => a + b, 0), L = TOP.cum[TOP.cum.length - 1], shift = u / CYC * run;
    for (let r = -2; r * run + shift < L; r++) {
      let s = r * run + shift;
      [ ...S ].forEach((ch, j) => {
        const c = s + ws[j] / 2;
        s += ws[j];
        if (c < 0 || c > L || ch === " ") return;
        const p = ptAt(TOP.pts, TOP.cum, c / L);
        k.save();
        k.translate(p[0], p[1]);
        k.rotate(p[2]);
        txt(k, ch, 0, 0, f, ch === "•" ? C.yel : "#fff", "center", "alphabetic");
        k.restore();
      });
    }
  };
