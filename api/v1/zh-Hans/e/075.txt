Motion Prompt #075 · 吸顶页头
公告条随滚动消失，但其下的页头停在顶边，内容从它下方继续滚动。

- 滚动量：0.4–1.3 s 为 0→110 px，1.9–2.8 s 为 110→250 px，3.4–4.4 s 为 250→0（cubic ease-in-out）。
- 最上方为 22 px 高的藏青色公告条，下方为 34 px 高的页头（标志、三个菜单项、按钮）。
- 页头位置 = max(22 − 滚动量, 0)：公告条消失后，页头停在顶边（position: sticky; top: 0）。
- 随后滚动 24 px 的过程中，页头下方的 12 px 阴影不透明度从 0 增至 0.16。
- 内容（照片、标题、三张卡片）在页头下方正常滚动。

Canvas: 539×208; cycle: 4.8s.

  // @element 03
    CELLS[2] = (k, w, h, u) => {
    frame(k);
    const y = steps([ [ .4, 0 ], [ 1.3, 110 ], [ 1.9, 110 ], [ 2.8, 250 ], [ 3.4, 250 ], [ 4.4, 0 ] ], u);
    const STRIP = 22;
    view(k, () => {
      k.save();
      k.translate(0, -y);
      pic(k, 24, 70, 214, 112, 0, 8);
      txt(k, MotionLocale.text("copy_081"), 262, 104, `800 22px ${FONT}`, C.ink);
      bars(k, 262, 120, 210, 3);
      btn(k, 262, 160, 100, 26, MotionLocale.text("copy_136"));
      k.fillStyle = C.soft;
      k.fillRect(0, 212, VW, 240);
      txt(k, MotionLocale.text("copy_051"), VW / 2, 246, `800 18px ${FONT}`, C.ink, "center");
      for (let i = 0; i < 3; i++) {
        const x = 34 + i * 152;
        k.fillStyle = "#fff";
        rr(k, x, 262, 136, 120, 8);
        k.fill();
        pic(k, x + 8, 270, 120, 58, i, 5);
        bars(k, x + 10, 340, 110, 3, C.faint, 6, 11);
      }
      k.fillStyle = C.ink;
      k.fillRect(0, 0, VW, STRIP);
      txt(k, MotionLocale.text("copy_161"), VW / 2, STRIP / 2 + .5, `600 11px ${FONT}`, "#fff", "center", "middle");
      k.restore();
      const hy = Math.max(STRIP - y, 0), sh = clamp((y - STRIP) / 24);
      if (sh > 0) {
        const g = k.createLinearGradient(0, hy + 34, 0, hy + 46);
        g.addColorStop(0, `rgba(28,35,51,${.16 * sh})`);
        g.addColorStop(1, "rgba(28,35,51,0)");
        k.fillStyle = g;
        k.fillRect(0, hy + 34, VW, 12);
      }
      header(k, hy);
      sbar(k, y / 250, .42);
    });
  };
