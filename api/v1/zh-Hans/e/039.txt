Motion Prompt #039 · 延时摄影
固定镜头将 06:00 到 18:00 压缩为几秒；太阳、云、人和影子快速运动。

- 相机：完全固定（高度 1.6 m，俯仰角 +0.08 rad，约向上 5°）。
- 时序：3.7 秒内流逝 12 小时（0.4–4.1 秒），三次方缓入缓出（约 cubic-bezier(0.65,0,0.35,1)）。
- 太阳：沿弧线从左向右移动，角度 a = −1.25 rad → +1.25 rad；纵深 170 m，位置为 (sin a×120, cos a×60 - 8) m。
- 三朵云总共向右漂移 140 m，每 80 m 循环；七个小人各自在宽 18 m 的范围内穿行 6 次（奇数和偶数编号方向相反）。
- 建筑的影子背向太阳，并随太阳运动而摆动（半透明灰色）。
- 返回：最后 0.5 秒交叉淡化回到起始状态；右上角时钟以 10 分钟为步长，从 06:00 走到 18:00。

    CELLS[14] = shot(camCell((k, w, h, p, u) => {
    const a = lerp(-1.25, 1.25, p);
    const clouds = [ [ -20 + mod(p * 140 + 0, 80) - 10, 18, 90, 2.2 ], [ -50 + mod(p * 140 + 40, 80), 24, 120, 3 ], [ -10 + mod(p * 140 + 20, 80) - 30, 13, 70, 1.6 ] ];
    const ex = [];
    for (let i = 0; i < 7; i++) {
      const ph = mod(p * 6 + i / 7, 1), dir = i % 2 ? 1 : -1;
      ex.push([ dir * lerp(-9, 9, ph), 0, 10 + i * 1.3, kk => {
        kk.fillStyle = i % 3 ? C.g4 : C.blueM;
        kk.fillRect(-.12, 0, .24, .9);
        kk.beginPath();
        kk.arc(0, 1.05, .13, 0, 7);
        kk.fill();
      } ]);
    }
    world(k, w, h, cam({
      y: 1.6,
      pitch: .08
    }), {
      sun: [ Math.sin(a) * 120, Math.cos(a) * 60 - 8, 170 ],
      sunCol: C.blueM,
      clouds: clouds,
      extra: ex,
      after: (kk, cc) => {
        kk.fillStyle = "rgba(160,160,165,.22)";
        BLD.slice(0, 6).forEach(([x, bw, bh]) => {
          const L = bh * .9;
          const q1 = proj(cc, x - bw / 2, 0, 48), q2 = proj(cc, x + bw / 2, 0, 48), q3 = proj(cc, x + bw / 2 - Math.sin(a) * L, 0, 48 - Math.cos(a) * L * .4 - 1), q4 = proj(cc, x - bw / 2 - Math.sin(a) * L, 0, 48 - Math.cos(a) * L * .4 - 1);
          kk.beginPath();
          kk.moveTo(q1.x, q1.y);
          kk.lineTo(q2.x, q2.y);
          kk.lineTo(q3.x, q3.y);
          kk.lineTo(q4.x, q4.y);
          kk.fill();
        });
      }
    });
  }, p => {
    const m = Math.round(lerp(6 * 60, 18 * 60, p) / 10) * 10;
    return `${String(Math.floor(m / 60)).padStart(2, "0")}:${String(m % 60).padStart(2, "0")}`;
  }, .4, 4.1));
