Motion Prompt #126 · 运动路径
纸飞机沿点线路径飞行，随路径转向，途中完成一个回环。

- 每 4.7 s 完成一趟，从屏幕左侧外飞到右侧外。路径为 x = 270 + 112θ − 150·sinθ，y = 106 − 74·cosθ，θ = −π→π，在顶部中央形成回环。
- θ 与时间成正比推进，因此沿路径的实际速度不恒定，回环看起来更慢。
- 飞机转向路径方向：角度 = atan2(74·sinθ, 112 − 150·cosθ)。
- 路径预先显示为点线，宽 6，虚线长度/间隔 2/15。后方跟随一条含 24 段的薄荷绿尾迹，每段 θ 长 0.03，越靠后越细、越淡。

Canvas: 539×208; cycle: 4.7s.

  // @element 06
    const MPX = th => 270 + 112 * th - 150 * Math.sin(th), MPY = th => 106 - 74 * Math.cos(th);
  const MP = (() => {
    const pts = [];
    for (let i = 0; i <= 400; i++) {
      const th = -Math.PI + 2 * Math.PI * i / 400;
      pts.push([ MPX(th), MPY(th) ]);
    }
    return pts;
  })();
  CELLS[5] = (k, w, h, u) => {
    stage(k);
    k.save();
    k.setLineDash([ 2, 15 ]);
    k.lineCap = "round";
    k.strokeStyle = C.dim;
    k.lineWidth = 6;
    k.beginPath();
    MP.forEach((p, i) => i ? k.lineTo(p[0], p[1]) : k.moveTo(p[0], p[1]));
    k.stroke();
    k.restore();
    const th = -Math.PI + 2 * Math.PI * (u / CYC);
    k.lineCap = "round";
    for (let j = 0; j < 24; j++) {
      const a = th - j * .03, b = th - (j + 1) * .03;
      k.strokeStyle = C.mint;
      k.globalAlpha = .9 * (1 - j / 24);
      k.lineWidth = 7 * (1 - j / 30);
      k.beginPath();
      k.moveTo(MPX(a), MPY(a));
      k.lineTo(MPX(b), MPY(b));
      k.stroke();
    }
    k.globalAlpha = 1;
    const x = MPX(th), y = MPY(th), ang = Math.atan2(74 * Math.sin(th), 112 - 150 * Math.cos(th));
    k.save();
    k.translate(x, y);
    k.rotate(ang);
    k.fillStyle = "#fff";
    k.beginPath();
    k.moveTo(32, 0);
    k.lineTo(-22, -22);
    k.lineTo(-12, 0);
    k.closePath();
    k.fill();
    k.fillStyle = "#C9D6FF";
    k.beginPath();
    k.moveTo(32, 0);
    k.lineTo(-22, 22);
    k.lineTo(-12, 0);
    k.closePath();
    k.fill();
    k.fillStyle = "#9FB3F0";
    k.beginPath();
    k.moveTo(32, 0);
    k.lineTo(-12, 0);
    k.lineTo(-6, 9);
    k.closePath();
    k.fill();
    k.restore();
  };
