Motion Prompt #101 · 伦勃朗光
侧上方各 45° 的光源亮起；脸部右侧进入阴影，只有阴影侧脸颊保留一个小的倒三角亮区。

- 场景 1：地平线位于 y 150，前方为肩部以上的人物（头部中心 x 270、y 78，半径 29 px，无面部或头发）。背景转为较暗场景，天空 #E7E5E1。
- 头部：x = 270 + 0.04·29 右侧为阴影 #33405A，左侧为亮部 #FFC75A。
- 亮三角顶点为 (x 270+0.24R, y 78−0.12R)、(x 270+0.7R, y 78−0.12R)、(x 270+0.24R, y 78+0.5R)，R = 29；约宽 13 px、高 18 px，与受光侧之间隔着约 6 px 宽的阴影带。
- 身体：中心 −0.45×56 px 左侧受光，+0.3×56 px 右侧为阴影；平面图显示光源位于 45°，右上角读数“45° · ↑45°”。
- 时序：效果在 0.4–1.4 s 渐入，保持到 4.0 s，并在 4.8 s 前消失；每轮 5 s，使用 cubic ease-in-out，约为 cubic-bezier(0.65,0,0.35,1)。

  // @element 05
    CELLS[4] = cell((k, u) => {
    const a = env(u), R = PR, W = torsoW(R);
    stagePortrait(k, pal(BASE, DIM, a));
    const lit = mixc(BASE.person, BASE.lit, a), sh = mixc(BASE.person, BASE.shade, a);
    bust(k, PX, PY, R, BASE.person, kk => {
      kk.fillStyle = lit;
      kk.beginPath();
      kk.arc(PX, PY, R, 0, TAU);
      kk.fill();
      kk.fillStyle = sh;
      kk.fillRect(PX + .04 * R, PY - R - 2, R * 2, R * 2 + 4);
      kk.beginPath();
      kk.arc(PX, PY, R, 0, TAU);
      kk.clip();
      kk.fillStyle = lit;
      kk.beginPath();
      kk.moveTo(PX + .24 * R, PY - .12 * R);
      kk.lineTo(PX + .7 * R, PY - .12 * R);
      kk.lineTo(PX + .24 * R, PY + .5 * R);
      kk.closePath();
      kk.fill();
    });
    bust(k, PX, PY, R, "rgba(0,0,0,0)", kk => {
      kk.fillStyle = "rgba(0,0,0,0)";
      torso(kk, PX, PY, R, PX - W, PX - .45 * W, lit);
      torso(kk, PX, PY, R, PX + .3 * W, PX + W, sh);
    });
    plan(k, [ [ Math.PI / 4, a ] ]);
    hud(k, "45° · ↑45°");
  });
