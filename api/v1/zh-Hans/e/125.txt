Motion Prompt #125 · 脉冲动画
爱心像心跳一样搏动，每次连跳两下，同时向外扩散环形轮廓。

- 珊瑚色爱心，每 1.175 s 一次心跳，每 4.7 s 四次；每次均为“咚哒”双跳。
- 缩放 = 1 + 0.26·G(0.2 s, 宽度 0.065) + 0.15·G(0.44 s, 宽度 0.07)，其中 G = e^(−((t − 中心) ÷ 宽度)²)；快速膨胀后立即回落。
- 两个心形轮廓分别从 0.16 s 和 0.40 s 开始扩散，在 0.8 s 内缩放 1.2→2.4，使用 cubic ease-out。
- 扩散时，不透明度 0.75→0，第二个轮廓按其 60% 计算；线宽 8.5→1.5 px。

  // @element 05
    CELLS[4] = (k, w, h, u) => {
    stage(k);
    const T = CYC / 4, lu = mod(u, T), X = w / 2, Y = h / 2 + 10;
    const g = (c, s) => Math.exp(-Math.pow((lu - c) / s, 2));
    const b = .26 * g(.2, .065) + .15 * g(.44, .07) + .26 * Math.exp(-Math.pow((lu - T - .2) / .065, 2));
    for (const [t0, s] of [ [ .16, 1 ], [ .4, .6 ] ]) {
      const q = P(lu, t0, t0 + .8);
      if (q <= 0 || q >= 1) continue;
      k.save();
      k.translate(X, Y);
      const sc = 1.2 + 1.2 * eOut(q);
      k.scale(sc, sc);
      heartPath(k, 44);
      k.strokeStyle = C.coral;
      k.globalAlpha = (1 - q) * .75 * s;
      k.lineWidth = (7 * (1 - q) + 1.5) / sc;
      k.stroke();
      k.restore();
    }
    k.globalAlpha = 1;
    k.save();
    k.translate(X, Y);
    k.scale(1 + b, 1 + b);
    heartPath(k, 44);
    k.fillStyle = C.coral;
    k.fill();
    k.fillStyle = "rgba(255,255,255,.75)";
    k.beginPath();
    k.ellipse(-30, -28, 9, 6, -.7, 0, 7);
    k.fill();
    k.restore();
  };
