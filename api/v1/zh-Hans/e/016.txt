Motion Prompt #016 · 视差
风景横向移动，近处图层移动更多、远处图层移动更少，形成纵深。

- 镜头横移：按正弦波在 ±108 px 范围内运动，每 4.7 秒往返一次。
- 每层沿相反方向偏移 pan × factor。系数从后向前为：星星 0.05、月亮 0.1、远山 0.28、中景山脊 0.6、前景山丘和树木 1.25、最近的深色形状 2.1。
- 远层使用接近背景的较暗颜色，近层更明亮；最近一层是沿底部铺开的大片深色剪影。
- 每层绘制范围至少超出画面左右边缘各 200 px，确保端部始终不可见。

Canvas: 539×208; cycle: 4.7s.

    const PLX = (() => {
    const R = rand(16), ridge = (base, amp, step, seed) => {
      const r = rand(seed), pts = [];
      for (let x = -220; x <= CW + 220; x += step) pts.push([ x, base - amp * (.35 + .65 * r()) ], [ x + step / 2, base + amp * .1 * r() ]);
      return pts;
    };
    const stars = [];
    for (let i = 0; i < 16; i++) stars.push([ R() * (CW + 80) - 40, 12 + R() * 80, 1.5 + R() * 2 ]);
    const trees = [];
    for (let i = 0; i < 9; i++) trees.push([ -160 + i * 108 + R() * 30, 34 + R() * 24, i % 3 ]);
    return {
      far: ridge(150, 78, 120, 3),
      mid: ridge(172, 74, 92, 8),
      stars: stars,
      trees: trees
    };
  })();
  CELLS[15] = (k, w, h, u) => {
    const pan = 108 * Math.sin(u / CYC * Math.PI * 2);
    const g = k.createLinearGradient(0, 0, 0, h);
    g.addColorStop(0, "#0F1A3D");
    g.addColorStop(1, "#27408F");
    k.fillStyle = g;
    k.fillRect(0, 0, w, h);
    k.fillStyle = "#fff";
    for (const s of PLX.stars) {
      k.globalAlpha = .8;
      k.beginPath();
      k.arc(s[0] - pan * .05, s[1], s[2], 0, 7);
      k.fill();
    }
    k.globalAlpha = 1;
    k.fillStyle = C.yel;
    k.beginPath();
    k.arc(390 - pan * .1, 56, 30, 0, 7);
    k.fill();
    const poly = (pts, m, col) => {
      k.fillStyle = col;
      k.beginPath();
      k.moveTo(-260, h + 10);
      pts.forEach(p => k.lineTo(p[0] - pan * m, p[1]));
      k.lineTo(w + 260, h + 10);
      k.closePath();
      k.fill();
    };
    poly(PLX.far, .28, "#31499C");
    poly(PLX.mid, .6, C.blue);
    k.fillStyle = C.mint;
    k.beginPath();
    k.moveTo(-300, h + 10);
    for (let x = -300; x <= w + 300; x += 20) k.lineTo(x - pan * 1.25, 176 + 12 * Math.sin(x * .012));
    k.lineTo(w + 300, h + 10);
    k.closePath();
    k.fill();
    for (const t of PLX.trees) {
      const x = t[0] - pan * 1.25, by = 180 + 12 * Math.sin(t[0] * .012), hh = t[1] * 1.45;
      k.fillStyle = C.navy;
      k.fillRect(x - 4, by - 16, 8, 22);
      k.fillStyle = t[2] === 0 ? C.coral : "#12A57B";
      k.beginPath();
      k.moveTo(x, by - hh);
      k.lineTo(x + 20, by - 12);
      k.lineTo(x - 20, by - 12);
      k.closePath();
      k.fill();
    }
    k.fillStyle = "#0A1230";
    [ [ -40, 60 ], [ 300, 44 ], [ 620, 66 ] ].forEach(([x, r]) => {
      k.beginPath();
      k.arc(x - pan * 2.1, h + 18, r, 0, 7);
      k.fill();
    });
  };
