Motion Prompt #135 · 万花筒
八向对称图案一边旋转一边从中心向外扩散，如同万花筒。

- 以中心为轴划分八个 45° 扇区，每隔一个扇区镜像绘制，形成八组镜像对称区域。
- 每个扇区包含 13 个图形（圆、菱形、三角形；四种颜色；固定随机种子）。图形从中心向外移动至半径 300，每 4.7 s 走完一个半径，到达边缘后回到中心。
- 图形越向外越大（(5 + 0.1r) × 0.7–1.4），每移动 1 px 旋转 0.02 rad。前 30 px 渐入，最后 70 px 渐出；各图形每 4.7 s 摆动一次，幅度 ±0.08 rad。
- 整个图案每 4.7 s 旋转 90°（两个扇区，保证循环衔接）。
- 中心为白色八角星，半径 20 ± 4，每 4.7 s 脉动两次。

Canvas: 539×208; cycle: 4.7s.

    for (let i = 0; i < 13; i++) a.push({

  CELLS[14] = (k, w, h, u) => {
    stage(k);
    const X = w / 2, Y = h / 2, RM = 300, rot = Math.PI / 2 * u / CYC, W8 = Math.PI / 4;
    for (let s = 0; s < 8; s++) {
      k.save();
      k.translate(X, Y);
      if (s % 2) {
        k.rotate(rot + (s + 1) * W8);
        k.scale(1, -1);
      } else k.rotate(rot + s * W8);
      k.beginPath();
      k.moveTo(0, 0);
      k.arc(0, 0, RM + 60, -.006, W8 + .006);
      k.closePath();
      k.clip();
      for (const it of KAL) {
        const r = mod(it.r0 + u / CYC * RM, RM), th = it.th + .08 * Math.sin(2 * Math.PI * u / CYC + it.ph), sz = (5 + r * .1) * it.sz;
        const a = smooth(P(r, 0, 30)) * (1 - smooth(P(r, RM - 70, RM)));
        if (a <= 0) continue;
        k.globalAlpha = a;
        k.fillStyle = it.col;
        k.save();
        k.translate(Math.cos(th) * r, Math.sin(th) * r);
        k.rotate(r * .02);
        if (it.kind === 0) {
          k.beginPath();
          k.arc(0, 0, sz, 0, 7);
          k.fill();
        } else if (it.kind === 1) {
          k.beginPath();
          k.moveTo(0, -sz * 1.2);
          k.lineTo(sz * .8, 0);
          k.lineTo(0, sz * 1.2);
          k.lineTo(-sz * .8, 0);
          k.closePath();
          k.fill();
        } else {
          k.beginPath();
          k.moveTo(0, -sz);
          k.lineTo(sz * .95, sz * .7);
          k.lineTo(-sz * .95, sz * .7);
          k.closePath();
          k.fill();
        }
        k.restore();
      }
      k.globalAlpha = 1;
      k.restore();
    }
    k.fillStyle = "#fff";
    star(k, X, Y, 20 + 4 * Math.sin(2 * Math.PI * 2 * u / CYC), rot, .5, 8);
    k.fill();
  };
