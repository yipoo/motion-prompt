Motion Prompt #013 · 液态动效
液态团块拉伸、分裂成液滴，再融合为一体。

- 领头圆（半径 44）：0.35–1.45 秒以 cubic-bezier(.7, 0, .2, 1) 向右移动 300 px，同时沿正弦弧线上升 34 px；2.7–3.8 秒以相同方式返回。
- 七个圆沿相同路径跟随，每个依次延迟 0.065 秒，半径从 44 逐渐减至 15，上下摆动 ±6 px。
- 即使停留时，三个小圆（半径 15、13、11）也会在离主体 34–64 px 的范围内进出，每周期两次。
- 融合效果：将圆绘为白色 → 模糊 9 px → 对 alpha 做阈值处理（仅在 112–144 之间保持柔和过渡）→ 用颜色填充保留区域。
- 填充为薄荷色 → 蓝色 → 紫色（#7C5CFF）的对角渐变，领头圆上有白色椭圆高光。

  // @element 13
    const LQ = bez(.7, 0, .2, 1);
  const lqPos = uu => {
    const u = mod(uu, CYC);
    let p;
    if (u < .35) p = 0; else if (u < 1.45) p = LQ(P(u, .35, 1.45)); else if (u < 2.7) p = 1; else if (u < 3.8) p = 1 - LQ(P(u, 2.7, 3.8)); else p = 0;
    const mvp = u >= .35 && u < 1.45 ? P(u, .35, 1.45) : u >= 2.7 && u < 3.8 ? P(u, 2.7, 3.8) : 0;
    return [ lerp(120, 420, p), 104 - 34 * Math.sin(Math.PI * mvp) ];
  };
  CELLS[12] = (k, w, h, u) => {
    stage(k);
    const blobs = [];
    for (let i = 0; i < 8; i++) {
      const [x, y] = lqPos(u - i * .065);
      blobs.push([ x, y + (i ? 6 * Math.sin(u * 2.67 + i) : 0), lerp(44, 15, i / 7) ]);
    }
    for (let j = 0; j < 3; j++) {
      const [x, y] = lqPos(u - .1), a = j * 2.2 + .6, d = 34 + 30 * (.5 - .5 * Math.cos(u / CYC * Math.PI * 4 + j * 2.1));
      blobs.push([ x + Math.cos(a) * d, y + Math.sin(a) * d * .8, 15 - j * 2 ]);
    }
    const L = goo(g => {
      for (const b of blobs) {
        g.beginPath();
        g.arc(b[0], b[1], b[2], 0, 7);
        g.fill();
      }
    }, 9, g => {
      const gr = g.createLinearGradient(60, 30, 480, 180);
      gr.addColorStop(0, C.mint);
      gr.addColorStop(.55, C.blue);
      gr.addColorStop(1, "#7C5CFF");
      g.fillStyle = gr;
      g.fillRect(0, 0, w, h);
    });
    k.drawImage(L, 0, 0);
    const [hx, hy] = lqPos(u);
    k.fillStyle = "rgba(255,255,255,.75)";
    k.beginPath();
    k.ellipse(hx - 14, hy - 17, 10, 6, -.6, 0, 7);
    k.fill();
  };
