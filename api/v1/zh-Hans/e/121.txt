Motion Prompt #121 · 预备动作
方形角色先逆着前进方向后撤、压扁蓄力，然后猛冲向前。

- 每 2.35 s 移动一次，每 4.7 s 两次：先从左向右，再从右向左。珊瑚色圆角方块，边长 86 px、圆角 24，带白色箭头；起点距边缘 108 px，地面 y = 178。
- 蓄力：0.2–0.95 s，quadratic ease-in-out。逆前进方向后撤 44 px，压扁至宽 1.22 倍 × 高 0.76 倍，并向后倾斜 0.18 rad。
- 发射：0.95–1.38 s，时长 0.43 s。从后撤位置以 backOut（s = 1.1）移至另一端，略微越过目标。前 60% 行程按半正弦拉伸，最大宽 1.55 倍 × 高 0.72 倍，并向前倾斜 0.08 rad。
- 落地：从 1.38 s 起，宽高以相反方向按 0.13·e^(−7t)·sin(17t) 晃动。
- 发射时，起点出现三团尘土，在行程前 70% 内消失，身后有三条宽 8 的黄色速度线。

Canvas: 539×208; cycle: 4.7s.

  // @element 01
    CELLS[0] = (k, w, h, u) => {
    u = snapU(u);
    stage(k);
    const T = CYC / 2, n = Math.floor(u / T), lu = u - n * T, dir = n % 2 ? -1 : 1;
    const XL = 108, XR = w - 108, xs = dir > 0 ? XL : XR, xe = dir > 0 ? XR : XL, floor = h - 30, S = 86, BACK = 44;
    k.fillStyle = C.navy3;
    rr(k, 26, floor, w - 52, 5, 2.5);
    k.fill();
    const wind = eIOs(P(lu, .2, .95)), go = P(lu, .95, 1.38);
    let x, sx = 1, sy = 1, rot = 0;
    if (go <= 0) {
      x = xs - dir * BACK * wind;
      sx = 1 + .22 * wind;
      sy = 1 - .24 * wind;
      rot = -dir * .18 * wind;
    } else {
      x = lerp(xs - dir * BACK, xe, backOut(go, 1.1));
      const st = go < .6 ? Math.sin(Math.PI * go / .6) : 0;
      sx = 1 + .55 * st;
      sy = 1 - .28 * st;
      rot = dir * .08 * st;
      const land = lu - 1.38;
      if (land > 0) {
        const wob = .13 * Math.exp(-7 * land) * Math.sin(17 * land);
        sy = 1 - wob;
        sx = 1 + wob;
      }
    }
    if (go > 0 && go < .7) {
      const q = go / .7;
      k.fillStyle = C.dim;
      k.globalAlpha = .8 * (1 - q);
      for (let i = 0; i < 3; i++) {
        k.beginPath();
        k.arc(xs - dir * (BACK + 30 + i * 22 + q * 30), floor - 8 - i * 6 - q * 10, 7 + q * 12 - i * 2, 0, 7);
        k.fill();
      }
      k.globalAlpha = 1;
    }
    k.fillStyle = "rgba(0,0,0,.3)";
    k.beginPath();
    k.ellipse(x, floor + 8, S * .62 * sx, 6, 0, 0, 7);
    k.fill();
    if (go > .02 && go < .8) {
      const a = 1 - go / .8;
      k.strokeStyle = C.yel;
      k.lineCap = "round";
      k.lineWidth = 8;
      k.globalAlpha = a;
      [ .28, .55, .8 ].forEach((f, i) => {
        const y = floor - S * sy * f, x0 = x - dir * (S * sx / 2 + 12), L = 120 * a + 20;
        k.beginPath();
        k.moveTo(x0, y);
        k.lineTo(x0 - dir * L * (i === 1 ? 1 : .7), y);
        k.stroke();
      });
      k.globalAlpha = 1;
    }
    k.save();
    k.translate(x, floor);
    k.rotate(rot);
    k.scale(sx, sy);
    k.fillStyle = C.coral;
    rr(k, -S / 2, -S, S, S, 24);
    k.fill();
    k.fillStyle = "rgba(255,255,255,.16)";
    rr(k, -S / 2, -S, S, S / 2, [ 24, 24, 0, 0 ]);
    k.fill();
    k.strokeStyle = "#fff";
    k.lineWidth = 10;
    k.lineCap = "round";
    k.lineJoin = "round";
    k.beginPath();
    k.moveTo(-dir * 12, -S / 2 - 20);
    k.lineTo(dir * 12, -S / 2);
    k.lineTo(-dir * 12, -S / 2 + 20);
    k.stroke();
    k.restore();
  };
