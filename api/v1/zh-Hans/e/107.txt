Motion Prompt #107 · 长曝光
街道变为夜景；行驶的汽车几乎隐去，红色尾灯光线在车后拉长，对向车道出现白色与琥珀色线条。

- 场景 2：街角；y 132–162 的道路后方有建筑、树木和路灯（x 114、292、494）；人物站在左前方，双脚位于 x 176、y 200，身高 124 px。配色转为夜景，天空 #2C3B5E，每三个窗户中一个为 #FFC75A。
- 汽车每 5 s 从 x −60→600；车身不透明度 100%→10%。
- 尾灯线从 x −60 延伸到汽车位置：y 145 处为 #FF5B4D、粗 2.6 px，y 149 处为 #FF8A6E、粗 1.4 px。
- 对向车道的线条横贯全宽：y 135 处 #FFE9B8、粗 2.6 px；y 138.5 处 #FFC75A、粗 1.6 px；y 141.5 处 #FFE9B8、粗 1.2 px；随效果淡入。
- 右上角快门速度 1/250→8″。
- 时序：效果在 0.4–1.4 s 渐入，保持到 4.0 s，并在 4.8 s 前消失；每轮 5 s，使用 cubic ease-in-out，约为 cubic-bezier(0.65,0,0.35,1)。

  // @element 11
    CELLS[10] = cell((k, u) => {
    const a = env(u), x = lerp(-60, 600, u / CYC);
    const p = pal(BASE, NIGHT, a);
    stageStreet(k, p, {
      car: false,
      person: false,
      after: D => {
        const kk = D.k;
        kk.save();
        kk.globalAlpha = a;
        kk.lineCap = "round";
        const x0 = -60, x1 = Math.min(x, CW + 40);
        const ln = (ya, yb, c1, c2, w1, w2) => {
          kk.strokeStyle = c1;
          kk.lineWidth = w1;
          kk.beginPath();
          kk.moveTo(x0, ya);
          kk.lineTo(x1, ya);
          kk.stroke();
          kk.strokeStyle = c2;
          kk.lineWidth = w2;
          kk.beginPath();
          kk.moveTo(x0, yb);
          kk.lineTo(x1, yb);
          kk.stroke();
        };
        ln(145, 149, "#FF5B4D", "#FF8A6E", 2.6, 1.4);
        for (const [y, c, lw] of [ [ 135, "#FFE9B8", 2.6 ], [ 138.5, BASE.lit, 1.6 ], [ 141.5, "#FFE9B8", 1.2 ] ]) {
          kk.strokeStyle = c;
          kk.lineWidth = lw;
          kk.beginPath();
          kk.moveTo(-10, y);
          kk.lineTo(CW + 10, y);
          kk.stroke();
        }
        kk.restore();
      }
    });
    k.save();
    k.globalAlpha = 1 - .9 * a;
    const D = pen(k);
    car(D, x, p, {
      head: BASE.lit,
      tail: "#FF5B4D"
    });
    k.restore();
    figure(pen(k), FIG[0], FIG[1], FIG[2], p.person);
    hud(k, a > .5 ? "8″" : "1/250");
  });
