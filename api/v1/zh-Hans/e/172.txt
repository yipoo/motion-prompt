Motion Prompt #172 · 长镜头
一个连续镜头跟随行走的人物，全程不切换；底部时间条始终连续不断。

- 衔接：无；一个镜头持续跟随人物 4.2 s（#041 Jump cut 会将其分成五段）。
- 主体：人物以恒速从左前方（x = −3.4 m，纵深 5.2 m）走向右后方（x = 2.8 m，纵深 9.6 m）；四肢按 sin(t×7)×0.42 rad 摆动。
- 相机：跟随人物向右横移 0→1.4 m，同时转动，偏航角 = atan2(person x - camera x, depth)×0.8；高度 1.4 m。
- 底部条带：一整条蓝色条，标注 一镜到底 · 无剪切（单镜头、无剪切），并带播放头，表明全程没有中断。
- 返回：4.2 s 停止，最后 0.5 s 交叉淡化回起点，一轮 4.7 s。
- 右上角显示已过时间：NO CUT 0.0s→4.2s。

  // @element 04
    CELLS[3] = shot((k, w, h, u) => {
    const uu = Math.min(u, CYC - RESET);
    const draw = (kk, t) => {
      const p = t / (CYC - RESET);
      const x = lerp(-3.4, 2.8, p), z = lerp(5.2, 9.6, p), cx = lerp(0, 1.4, p);
      world(kk, w, h, cam({
        x: cx,
        y: 1.4,
        yaw: Math.atan2(x - cx, z) * .8
      }), {
        person: {
          x: x,
          z: z,
          walk: t * 7
        },
        fg: [ [ -2.6, 3.4 ], [ 3.6, 4.6, "rock" ] ]
      });
      strip(kk, w, h, [ [ 0, CYC, MotionLocale.text("copy_055"), C.blue ] ], t);
    };
    const r = P(u, CYC - RESET, CYC);
    blend(k, kk => draw(kk, uu), kk => draw(kk, 0), eIOs(r));
    return `NO CUT ${(r > .5 ? 0 : uu).toFixed(1)}s`;
  });
