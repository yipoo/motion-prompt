Motion Prompt #120 · 双重曝光
背景变白，山丘、房屋、树木和太阳构成的风景缩小后，仅显示在半身像轮廓内部。

- 场景 3：左侧为半身像（头部中心 x 140、y 82，半径 28 px），右侧有山丘、房屋、树木和太阳（x 432、y 54，半径 15 px）。
- 背景：半身像外部为白色 #FFFFFF。
- 半身像内部：同一风景缩放至 0.7 倍，摆放位置使太阳位于头部右上方，山脊横穿颈部。
- 内部颜色：天空 #D3DBE6，山丘 #9AA8BD 和 #76869F，地面 #55647E，树木 #3F4C66。
- 过渡：按效果强度将双重曝光图叠加到原图上。
- 时序：效果在 0.4–1.4 s 渐入，保持到 4.0 s，并在 4.8 s 前消失；每轮 5 s，使用 cubic ease-in-out，约为 cubic-bezier(0.65,0,0.35,1)。

Canvas: 539×208; cycle: 5s.

  // @element 24
    CELLS[23] = cell((k, u) => {
    const a = env(u);
    const DX = Object.assign({}, BASE, {
      sky: "#D3DBE6",
      sky2: "#D3DBE6",
      far: "#9AA8BD",
      mid: "#76869F",
      ground: "#55647E",
      bld: "#E8ECF2",
      edge: "#55647E",
      win: "#9AA8BD",
      winOff: "#9AA8BD",
      tree: "#3F4C66",
      trunk: "#3F4C66"
    });
    const draw = kk => {
      kk.fillStyle = "#FFFFFF";
      kk.fillRect(0, 0, CW, CH);
      kk.save();
      bustPath(kk, QX, QY, QR);
      kk.clip();
      kk.translate(QX, QY);
      kk.scale(.7, .7);
      kk.translate(-415, -77);
      kk.fillStyle = DX.sky;
      kk.fillRect(0, -200, CW * 2, CH * 3);
      stageLand(kk, DX, {
        person: false
      });
      kk.fillStyle = DX.ground;
      kk.fillRect(-CW, 200, CW * 3, CH * 3);
      kk.restore();
    };
    if (a >= .999) {
      draw(k);
      return;
    }
    stageLand(k, BASE);
    if (a < .002) return;
    const [L, kl] = layer("dx");
    draw(kl);
    k.save();
    k.globalAlpha = a;
    k.drawImage(L, 0, 0);
    k.restore();
  });
