Motion Prompt #038 · 速度渐变
同一个镜头内，播放速度从快平滑变慢；左下角曲线图显示速度。

- 相机：完全固定（高度 1.3 m）；仅播放速度变化。
- 播放速度：2.4× → 0.25×；变化持续 0.7 秒（1.2–1.9 秒），二次方缓入缓出（约 cubic-bezier(0.45,0,0.55,1)）。
- 主体：8 m 外的人物从左侧 6 m 处向右跑；x = -6 + 素材时间 × 2.0 m；快速阶段迅速接近。
- 跳跃：x 在 −1.5 m 至 2.5 m 之间时沿抛物线跳起，最高 0.99 m，双臂举起；约 0.94 秒起跳，大部分腾空过程以慢动作播放，约 3.68 秒落地。
- 曲线图：左下角 120×40 px 的速度曲线，蓝点（半径 5 px）标记当前时间。
- 返回：4.2 秒停止，最后 0.5 秒交叉淡化返回；右上角显示速率（×2.4 → ×0.3）。

  // @element 14
    function rampTime(u) {
    const v = x => lerp(2.4, .25, eIOs(P(x, 1.2, 1.9)));
    let s = 0;
    const N = 60;
    for (let i = 0; i < N; i++) {
      const x = u * (i + .5) / N;
      s += v(x) * u / N;
    }
    return {
      s: s,
      v: v(u)
    };
  }
  CELLS[13] = shot((k, w, h, u) => {
    const uu = Math.min(u, CYC - RESET);
    const draw = (kk, uu) => {
      const {s: s, v: v} = rampTime(uu);
      const x = -6 + s * 2, q = P(x, -1.5, 2.5);
      world(kk, w, h, cam({
        y: 1.3
      }), {
        person: {
          x: x,
          z: 8,
          walk: s * 9,
          y: jumpY(q) * .9,
          armUp: q > 0 && q < 1
        },
        after: k2 => {
          const gx = -w / 2 + 26, gy = h / 2 - 22, gw = 120, gh = 40;
          k2.fillStyle = "rgba(255,255,255,.85)";
          rr(k2, gx - 8, gy - gh - 10, gw + 16, gh + 18, 6);
          k2.fill();
          k2.strokeStyle = C.faint;
          k2.lineWidth = 1.5;
          k2.beginPath();
          k2.moveTo(gx, gy);
          k2.lineTo(gx + gw, gy);
          k2.stroke();
          k2.strokeStyle = C.blue;
          k2.lineWidth = 2.5;
          k2.beginPath();
          for (let i = 0; i <= 40; i++) {
            const xx = i / 40 * (CYC - RESET), vv = rampTime(xx).v;
            const X = gx + i / 40 * gw, Y = gy - vv / 2.4 * gh;
            i ? k2.lineTo(X, Y) : k2.moveTo(X, Y);
          }
          k2.stroke();
          const X = gx + uu / (CYC - RESET) * gw, Y = gy - v / 2.4 * gh;
          k2.fillStyle = C.blue;
          k2.beginPath();
          k2.arc(X, Y, 5, 0, 7);
          k2.fill();
        }
      });
      return v;
    };
    const r = P(u, CYC - RESET, CYC);
    let v;
    blend(k, kk => {
      v = draw(kk, uu);
    }, kk => draw(kk, 0), eIOs(r));
    return `×${(r > .5 ? 2.4 : v).toFixed(1)}`;
  });
