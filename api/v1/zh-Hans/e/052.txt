Motion Prompt #052 · 折叠面板
点击三个标题中的第一项，其内容展开并把下方行向下推；再次点击则折叠。

- 542×191 卡片上的 4.5 秒周期。三个标题（宽 494、高 34、圆角半径 8、间距 6），右侧各有向下折角箭头。
- 光标在 0.95 秒到达第一个标题，于 1.0 秒和 3.3 秒点击。
- 展开程度 ex：1.0–1.35 秒以 easeInOutCubic 从 0 变为 1，随后 3.3–3.62 秒从 1 变为 0。
- 第一行从高 34 增至 92（+58·ex），把下方行向下推；两行内容裁剪在行内，以不透明度 ex 淡入。
- 展开时，折角箭头旋转 180°·ex，边框变浅蓝，标题变蓝。

  // @element 04
    CELLS[3] = ui((w, h, t) => {
    const ex = eio(seg(t, 1, 1.35)) * (1 - eio(seg(t, 3.3, 3.62)));
    let y = 12;
    const H0 = 34, EX = 58;
    for (let i = 0; i < 3; i++) {
      const extra = i === 0 ? EX * ex : 0;
      strokeRR(24, y, 494, H0 + extra, 8, i === 0 && ex > .05 ? C.sb : C.line2, 1.5);
      bar(42, y + 12, [ 180, 220, 150 ][i], i === 0 && ex > .05 ? C.blue : C.line2, 10);
      chevron(494, y + 17, 6, i === 0 ? Math.PI * ex : 0, i === 0 && ex > .05 ? C.blue : "#9a9896");
      if (i === 0 && ex > 0) {
        X.save();
        rr(24, y, 494, H0 + extra, 8);
        X.clip();
        X.globalAlpha = ex;
        bar(42, y + H0 + 6, 420);
        bar(42, y + H0 + 24, 320);
        X.restore();
      }
      y += H0 + extra + 6;
    }
    pointer([ [ .3, w + 30, h + 30 ], [ .95, 300, 31 ], [ 1.6, 300, 31 ], [ 3, 300, 31 ], [ 3.55, 300, 31 ], [ 4.2, w + 30, h + 30 ] ], [ 1, 3.3 ], t);
  });
