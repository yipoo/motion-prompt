Motion Prompt #139 · 立方体转场
如同立方体转动，正面转离视线，下一面随之转入。

- A 面为珊瑚色背景上的 A，B 面为薄荷色背景上的 B，各为 250 × 146。每段 2.35 s，A→B→A。
- 0.45–1.6 s（持续 1.15 s）旋转 0→90°，三次缓入缓出，绕立方体中心的竖直轴旋转（深度等于面宽 250）。
- 透视：相机距离 820 px。每个面按 2 px 宽的竖向切片绘制，各切片的大小与位置由透视系数决定。
- 转出的面最多变暗 55%（0.55·sinφ）；转入的面从变暗 55% 开始逐渐变亮（0.55·cosφ）。地面阴影随可见宽度变化。

Canvas: 539×208; cycle: 4.7s.

    const CFW = 250, CFH = 146, CF = [ mkCanvas(CFW, CFH), mkCanvas(CFW, CFH) ];
  function cubeFace(i) {
    const g = CF[i].getContext("2d");
    g.setTransform(1, 0, 0, 1, 0, 0);
    g.clearRect(0, 0, CFW, CFH);
    g.fillStyle = i ? C.mint : C.coral;
    g.fillRect(0, 0, CFW, CFH);
    g.strokeStyle = "rgba(255,255,255,.35)";
    g.lineWidth = 4;
    g.strokeRect(12, 12, CFW - 24, CFH - 24);
    txt(g, i ? "B" : "A", 78, CFH / 2 + 4, `800 104px ${DISP}`, i ? C.navy : "#fff", "center", "middle");
    g.fillStyle = i ? C.navy : "#fff";
    g.globalAlpha = .8;
    rr(g, 140, 46, 78, 14, 7);
    g.fill();
    g.globalAlpha = .45;
    rr(g, 140, 72, 60, 12, 6);
    g.fill();
    rr(g, 140, 94, 70, 12, 6);
    g.fill();
    g.globalAlpha = 1;
    return CF[i];
  }
  CELLS[18] = abCell((k, w, h, u, a, b) => {
    stage(k);
    const phi = Math.PI / 2 * eIO(P(u, TA - .05, TB + .1)), c = Math.cos(phi), s = Math.sin(phi), HW = CFW / 2, D = 820, X = w / 2, Y = h / 2 - 4;
    const rot = (x, z) => [ x * c - z * s, x * s + z * c ];
    const FL = rot(-HW, HW), FR = rot(HW, HW), BR = rot(HW, -HW);
    const pj = p => {
      const f = D / (D - (p[1] - HW));
      return [ X + p[0] * f, f ];
    };
    k.fillStyle = "rgba(0,0,0,.3)";
    k.beginPath();
    k.ellipse(X, h - 16, HW * (c + s) * .95, 7, 0, 0, 7);
    k.fill();
    const face = (img, P0, P1, shade) => {
      const A0 = pj(P0), A1 = pj(P1);
      if (A1[0] - A0[0] < 1) return;
      const STEP = 2;
      for (let sx = 0; sx < CFW; sx += STEP) {
        const t0 = sx / CFW, t1 = Math.min(1, (sx + STEP) / CFW);
        const q0 = pj([ lerp(P0[0], P1[0], t0), lerp(P0[1], P1[1], t0) ]), q1 = pj([ lerp(P0[0], P1[0], t1), lerp(P0[1], P1[1], t1) ]);
        const hh = CFH * q0[1];
        k.drawImage(img, sx, 0, STEP, CFH, q0[0], Y - hh / 2, q1[0] - q0[0] + .7, hh);
      }
      if (shade > .01) {
        k.fillStyle = `rgba(8,14,40,${shade})`;
        k.beginPath();
        k.moveTo(A0[0], Y - CFH * A0[1] / 2);
        k.lineTo(A1[0], Y - CFH * A1[1] / 2);
        k.lineTo(A1[0], Y + CFH * A1[1] / 2);
        k.lineTo(A0[0], Y + CFH * A0[1] / 2);
        k.closePath();
        k.fill();
      }
    };
    face(cubeFace(a), FL, FR, .55 * s);
    face(cubeFace(b), FR, BR, .55 * c);
  });
