Motion Prompt #147 · 堆叠卡片
滚动时，下一张卡片从下方升起，略微错位地叠在前一张上；下层卡片稍微缩小并变暗。

- 视口 511 × 168。滚动步进 s：0.3–1.1 s 从 0→1，1.6–2.4 s 从 1→2，3.2–4.4 s 从 2→0（三次缓入缓出）。一轮 4.7 s（每个 9.4 s 的效果包周期两轮）。
- 三张卡片尺寸 371 × 124，圆角半径 12，居中排列，在页面上相隔 170 px；s 每增加一步，卡片上移 170 px。
- 卡片分别停在距顶部 14、26、38 px 的位置（position: sticky，每张卡片的 top 偏移比上一张大 12 px），使新卡片覆盖前一张。
- 上方每叠一张卡片，下层卡片就以顶边中点为中心缩小 4.5%（两张则 9%），并由藏青色覆盖层压暗，最多 16%。
- 卡片阴影：模糊 14 px，向上偏移 2 px。滚动条滑块随滚动移动。

Canvas: 539×208; cycle: 9.4s.

  // @element 03
    CELLS[2] = (k, w, h, u) => {
    u = snapU(u);
    frame(k);
    const s = steps([ [ .3, 0 ], [ 1.1, 1 ], [ 1.6, 1 ], [ 2.4, 2 ], [ 3.2, 2 ], [ 4.4, 0 ] ], u);
    const CARDS = [ [ C.blueL, C.ink, C.acc, "01", MotionLocale.text("copy_085"), 0 ], [ C.ink, "#fff", C.acc, "02", MotionLocale.text("copy_082"), 2 ], [ C.acc, "#fff", "#fff", "03", MotionLocale.text("copy_083"), 1 ] ];
    const X = 70, CWD = 371, CHT = 124, STEP = 170;
    view(k, () => {
      k.fillStyle = C.soft;
      k.fillRect(0, 0, VW, VH);
      CARDS.forEach(([bg, fg, nc, no, ttl, pv], i) => {
        const top = Math.max(14 + i * 12, 16 + i * STEP - s * STEP);
        const cov = clamp(s - i), sc = 1 - .045 * cov - .045 * clamp(s - i - 1);
        k.save();
        k.translate(VW / 2, top);
        k.scale(sc, sc);
        k.translate(-VW / 2, 0);
        k.save();
        k.shadowColor = "rgba(28,35,51,.22)";
        k.shadowBlur = 14;
        k.shadowOffsetY = -2;
        k.fillStyle = bg;
        rr(k, X, 0, CWD, CHT, 12);
        k.fill();
        k.restore();
        txt(k, no, X + 24, 44, `800 26px ${EN}`, nc);
        txt(k, ttl, X + 24, 78, `800 20px ${FONT}`, fg);
        bars(k, X + 24, 92, 140, 2, i === 0 ? "#C3CDEA" : "rgba(255,255,255,.35)");
        pic(k, X + CWD - 160, 14, 144, CHT - 28, pv, 8);
        if (cov > 0) {
          k.fillStyle = `rgba(28,35,51,${.16 * cov})`;
          rr(k, X, 0, CWD, CHT, 12);
          k.fill();
        }
        k.restore();
      });
      sbar(k, s / 2, .4);
    });
  };
