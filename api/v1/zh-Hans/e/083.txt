Motion Prompt #083 · 光标跟随
光标在页面上移动时，一个橙色圆会稍稍滞后地跟随它。

- 光标路径：x = 255 + 175·sin(2πt/4.8)，y = 86 + 46·sin(4πt/4.8)，构成横置的 8 字形，每 4.8 s 一圈。
- 圆形半径 17、橙色、不透明度 0.88，绘制在光标下方。
- 跟随规则：每帧（60 fps）走完剩余距离的 12%（x += (光标 x − x) × 0.12）。
- 箭头光标保持可见。

Canvas: 539×208; cycle: 4.8s.

  // @element 11
    CELLS[10] = (k, w, h, u) => {
    frame(k);
    const cur = t => [ 255 + 175 * Math.sin(2 * Math.PI * t / CYC), 86 + 46 * Math.sin(4 * Math.PI * t / CYC) ];
    let t = u - 2, [fx, fy] = cur(t);
    for (let i = 0; i < 120; i++) {
      t += 1 / 60;
      const [tx, ty] = cur(t);
      fx += (tx - fx) * .12;
      fy += (ty - fy) * .12;
    }
    view(k, () => {
      txt(k, MotionLocale.text("CREATIVE STUDIO"), VW / 2, 74, `800 34px ${EN}`, C.ink, "center", "alphabetic", "3px");
      bars(k, VW / 2 - 120, 96, 240, 2, C.faint, 6, 12, .7);
      btn(k, VW / 2 - 50, 128, 100, 24, MotionLocale.text("copy_135"), C.ink);
      k.fillStyle = C.acc;
      k.globalAlpha = .88;
      k.beginPath();
      k.arc(fx, fy, 17, 0, 7);
      k.fill();
      k.globalAlpha = 1;
      const [cx, cy] = cur(u);
      cursor(k, cx, cy);
    });
  };
